1     /* Dsp.cpp -- Front-end DSP primitives feeding the burst detector.
2      *
3      * Two single-pole IIR filters live here, both designed for streaming
4      * use (tiny per-sample state, no buffering, no look-ahead):
5      *
6      *   tg_hpf : single-pole high-pass / DC blocker. Removes microphone
7      *            wind, table rumble, and AC mains hum that would otherwise
8      *            bias the envelope follower below. Default cutoff 200 Hz
9      *            sits above most low-frequency noise and well below the
10      *            spectral content of the watch ticks themselves (which
11      *            peak in the 1-5 kHz range).
12      *
13      *            Implementation: y[n] = a * (y[n-1] + x[n] - x[n-1])
14      *            where a = exp(-2*pi*fc/fs). This is the canonical
15      *            DC-blocker form; pole at a (just inside unit circle on
16      *            the real axis), zero at +1.
17      *
18      *   tg_envelope : full-wave rectifier followed by a single-pole
19      *                 low-pass smoother. The output approximates the
20      *                 instantaneous envelope of the signal -- positive,
21      *                 slow-moving, with sharp rises on impulse onsets.
22      *
23      *                 Default smoothing time constant 0.15 ms. This is
24      *                 short enough that A and C clusters (which are
25      *                 milliseconds apart even on the fastest watches) stay
26      *                 well-separated in the envelope, but long enough to
27      *                 collapse the high-frequency carrier inside each
28      *                 impulse into a single hump for the detector to
29      *                 measure.
30      *
31      *                 Implementation: env[n] = env[n-1] + alpha * (|x[n]| - env[n-1])
32      *                 where alpha = 1 - exp(-1/(tau*fs)).
33      *
34      * Neither block is sample-rate-dependent in its API; everything is
35      * specified in seconds / Hz and computed against the configured fs at
36      * init time.
37      *
38      * V5.5: file renamed from dsp.c to Dsp.cpp; compiles under C++17.
39      * No algorithmic change.
40      */
41     #include "Dsp.h"

44     #ifndef M_PI
45     #define M_PI 3.14159265358979323846
46     #endif
47     
48     /* ---------- single-pole highpass (DC blocker) ---------- */
49     
50     void tg_hpf_init(tg_hpf_t *h, double fs, double f_c) {
51         if (f_c < 1.0)           f_c = 1.0;
52         if (f_c > 0.25 * fs)     f_c = 0.25 * fs;
53         h->a = exp(-2.0 * M_PI * f_c / fs);
54         tg_hpf_reset(h);
55     }
56     
57     void tg_hpf_reset(tg_hpf_t *h) {
58         h->x_prev = 0.0;
59         h->y_prev = 0.0;
60     }
61     
62     void tg_hpf_process(tg_hpf_t *h, const float *in, float *out, size_t n) {
63         double a  = h->a;
64         double xp = h->x_prev;
65         double yp = h->y_prev;
66         for (size_t i = 0; i < n; ++i) {
67             double x = (double)in[i];
68             double y = x - xp + a * yp;
69             out[i]   = (float)y;
70             xp = x;
71             yp = y;
72         }
73         h->x_prev = xp;
74         h->y_prev = yp;
75     }
76     
77     /* ---------- envelope detector ---------- */
78     
79     void tg_envelope_init(tg_envelope_t *e, double fs, double smoothing_ms) {
80         if (smoothing_ms <= 0.0) smoothing_ms = 0.15;
81         double tau_samples = (smoothing_ms * 1e-3) * fs;
82         if (tau_samples < 1.0) tau_samples = 1.0;
83         e->alpha = 1.0 - exp(-1.0 / tau_samples);
84         tg_envelope_reset(e);
85     }
86     
87     void tg_envelope_reset(tg_envelope_t *e) {
88         e->state = 0.0;
89     }
90     
91     void tg_envelope_process(tg_envelope_t *e, const float *in, float *out, size_t n) {
92         double s = e->state;
93         double a = e->alpha;
94         for (size_t i = 0; i < n; ++i) {
95             double x = (double)in[i];
96             if (x < 0.0) x = -x;
97             s += a * (x - s);
98             out[i] = (float)s;
99         }
100         e->state = s;
101     }
