4     /*
5         watch_synth_stream.h
6     
7         Continuous mono float-PCM synthetic mechanical-watch stream generator.
8     
9         -------------------------------------------------------------------------
10         PURPOSE
11         -------------------------------------------------------------------------
12         This module synthesizes a continuous stream of mechanical-watch-like sound.
13         It is intended for detector and timegrapher display testing where known
14         ground-truth timing is valuable.
15     
16         The generator is stateful. The caller repeatedly provides a destination
17         float buffer and the module fills that buffer with the next contiguous block
18         of samples. There is no internal output file dependency and no assumption
19         that the caller's block size is constant.
20     
21             WatchSynthStreamFillResult r = watch_synth_stream_fill_f32(
22                 &stream,
23                 out_pcm,       // caller-owned float buffer
24                 out_count,     // number of samples requested
25                 events,        // optional ground-truth event side channel
26                 event_capacity);
27     
28         -------------------------------------------------------------------------
29         IMPORTANT UNITS AND NAMING
30         -------------------------------------------------------------------------
31         pcm_peak_amplitude
32             Unit: normalized float PCM full-scale fraction.
33             Range: 0.0..1.0.
34             Example: 0.70 means the generated digital audio impacts are roughly
35             +/-0.70 full scale before noise/variation/clipping.
36     
37             This is digital loudness only. It is NOT the mechanical watch amplitude.
38     
39         watch_amplitude_degrees
40             Unit: degrees.
41             Meaning: simulated balance amplitude as a timegrapher would report.
42             Typical examples: 180, 220, 270, 300 degrees.
43     
44         lift_angle_degrees
45             Unit: degrees.
46             Meaning: watch lift angle used with watch_amplitude_degrees to derive
47             the A-to-C time. Typical values are often around 44..60 degrees.
48     
49         beat_error_ms
50             Unit: milliseconds, double precision.
51             Sign convention:
52     
53                 beat_error_ms = 0.5 * (interval(Tick -> Tock) - interval(Tock -> Tick))
54     
55             Therefore +0.210 means Tick->Tock is 0.420 ms longer than Tock->Tick, and a timegrapher should display about 0.210 ms.
56     
57         timing_jitter_us and bph_wander_*_us
58             Unit: microseconds.
59             These are intentionally kept in microseconds because they are tiny
60             perturbations around the nominal beat interval.
61     
62             Realistic defaults are deliberately small so rate_error_s_per_day remains
63             visible and accurate in short tests. Large wander values intentionally
64             make the displayed rate drift above or below the configured rate.
65     
66         -------------------------------------------------------------------------
67         A-TO-C TIME / WATCH AMPLITUDE MODEL
68         -------------------------------------------------------------------------
69         If use_watch_amplitude_for_a_to_c is enabled, A-to-C timing is derived from
70         BPH, lift angle, and mechanical watch amplitude:
71     
72             beat_interval_s = 3600 / BPH
73     
74             A_to_C_s =
75                 (2 * beat_interval_s / pi) *
76                 asin(lift_angle_degrees / (2 * watch_amplitude_degrees))
77     
78         Why this shape:
79             A watch balance is approximated as sinusoidal around center.
80             The lift angle is swept around the zero-crossing / impulse zone.
81             If the balance amplitude is A and lift angle is L, the time spent
82             crossing the lift angle is proportional to asin(L / (2A)), not simply
83             L / A over the whole beat interval.
84     
85         Larger watch amplitude produces a shorter A->C time; smaller watch
86         amplitude produces a longer A->C time.
87     
88         This is still a simplified synthetic timing model, but it is much closer
89         to the usual timegrapher geometry than the older linear formula. The older
90         formula was too large by roughly pi for typical amplitudes.
91     
92         -------------------------------------------------------------------------
93         CLEAN VS REALISTIC DEFAULTS
94         -------------------------------------------------------------------------
95         watch_synth_stream_clean_config()
96             Produces stable, repeatable packets with minimal variation. Useful for
97             unit tests and formula verification.
98     
99         watch_synth_stream_realistic_config()
100             Enables packet-shape variation, amplitude drift, contact/case
101             resonance, band-limited noise, BPH wander, and Tick/Tock spectral
102             differences. Useful for stress testing real detectors.
103     
104         watch_synth_stream_default_config()
105             Alias for the realistic configuration.
106     
107         -------------------------------------------------------------------------
108         STREAMING CONTRACT
109         -------------------------------------------------------------------------
110         - The stream starts at absolute sample index 0 after init/reset.
111         - Each fill call continues exactly where the previous fill call stopped.
112         - Any caller block size is valid, including odd sizes such as 257 samples.
113         - Packets may begin in one block and continue into later blocks.
114         - Resonator and noise filter states persist across blocks.
115         - Returned events are optional. Passing events=NULL disables the side channel
116           but does not change the generated audio.
117     
118         Build example:
119           gcc -O2 -std=c99 -Wall -Wextra -pedantic \
120               watch_synth_stream.c example_stream_to_float.c -lm -o example_stream
121     */

156         double max_a_to_c_time_s;          /* seconds. Safety clamp for A->C. */

161         int enable_sensor_resonance;       /* 1 = contact/case resonator model. */
162         int enable_bandlimited_noise;      /* 1 = band-limit synthetic mechanical noise. */
163         int enable_bph_wander;             /* 1 = tiny random-walk interval wander; keep small for rate-accurate tests. */
164         int enable_tick_tock_spectral_diff;/* 1 = Tick and Tock differ spectrally. */

166         /*
167             C-peak control. The exact C anchor is a narrow Gaussian peak placed at

178         double packet_gain_variation;      /* fraction. Per-packet random PCM gain variation. */
179         double shape_delay_jitter_us;      /* us. Random lobe delay perturbation. */
180         double shape_frequency_jitter;     /* fraction. Random lobe frequency variation. */
181         double shape_decay_jitter;         /* fraction. Random lobe decay variation. */
182     
183         double amplitude_drift_depth;      /* fraction. Example 0.08 = +/-8%. */
184         double amplitude_drift_period_s;   /* seconds. */
185         double bph_wander_depth_us;        /* us. Random-walk clamp. Large values intentionally disturb displayed rate. */
186         double bph_wander_step_us;         /* us. Random-walk step per beat. */
187     
188         double sensor_resonance1_hz;       /* Hz. */
189         double sensor_resonance1_q;        /* unitless Q. */
190         double sensor_resonance1_gain;     /* mix gain. */
191         double sensor_resonance2_hz;       /* Hz. */
192         double sensor_resonance2_q;        /* unitless Q. */
193         double sensor_resonance2_gain;     /* mix gain. */
194     
195         double noise_low_hz;               /* Hz. Noise high-pass corner. */
196         double noise_high_hz;              /* Hz. Noise low-pass corner. */
