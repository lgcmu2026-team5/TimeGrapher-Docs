51     void TSimWorker::StartSim(WatchSynthStreamConfig cfg)
52     {
53         int                        BytesIn;
54         double                     CurrentTime;
55         qint64                     Start,Delta,SleepTime;
56         char                       err[256];
57         WatchSynthStream           stream;
58         WatchSynthStreamEvent      events[16];
59         WatchSynthStreamFillResult r;
60         cfg.sample_rate_hz=mSamplesPerSecond;
61     
62         if (!watch_synth_stream_init(&stream, &cfg, err, sizeof(err)))
63         {
64             fprintf(stderr, "init failed: %s\n", err);
65             emit SimDone();
66             emit finished();
67             return;
68         }
69     
70         if (!mTimerStarted)
71         {
72             mTimerStarted=true;
73             mTimer.start();
74         }
75     
76         while (1)
77         {
78             Start=mTimer.elapsed();
79     
80             r = watch_synth_stream_fill_f32(&stream,  (float *)mDataIn, mDataInSize, events, 16);
81             if (r.samples_written != mDataInSize) {
82                 fprintf(stderr, "short fill\n");
83                 break;
84             }
85             if (QThread::currentThread()->isInterruptionRequested())
86             {
87                 break; // Exit loop early
88             }
89             unsigned int NumberOfSamples=r.samples_written;
90     
91             mRawAudio->Mutex.lock();
92             unsigned int TempWriteIndex = mRawAudio->WriteIndex;
93             mRawAudio->Mutex.unlock();
94             int SamplesLeft=std::min(NumberOfSamples,mRawAudio->NumberOfAudioSamples-TempWriteIndex);
95             memcpy(&mRawAudio->Samples[TempWriteIndex], mDataIn, SamplesLeft * SAMPLE_SIZE);
96             if(SamplesLeft < NumberOfSamples)
97             {
98                 memcpy(mRawAudio->Samples, &mDataIn[SamplesLeft], (NumberOfSamples - SamplesLeft) * SAMPLE_SIZE);
99                 qInfo() << "MasterPlaybackData Samples Rollover";
100             }
101             mRawAudio->Mutex.lock();
102             mRawAudio->WriteIndex = (TempWriteIndex+ NumberOfSamples) %  mRawAudio->NumberOfAudioSamples;
103             mRawAudio->TotalSamplesWritten+=NumberOfSamples;
104             mRawAudio->Mutex.unlock();
105             emit SimDataReady(); // Emit data to the main thread
106     
107             ++mFrameCount;
108             mSampleCount+=NumberOfSamples;
109             CurrentTime = mTimer.elapsed()/1000.0;
110             if (CurrentTime-mLastTime > 2) // average fps over 2 seconds
111             {
112                 double fdelta;
113                 fdelta=CurrentTime-mLastTime;
114                 mRawAudio->FPS=mFrameCount/fdelta;
115                 mRawAudio->SPS=mSampleCount/fdelta;
116                 mRawAudio->SPF=mSampleCount/mFrameCount;
117                 mLastTime=CurrentTime;
118                 mFrameCount=0;
119                 mSampleCount=0;
120             }
121             Delta=(mTimer.elapsed()-Start)+DELAY_FUGE_TIME_MS;
122             SleepTime=SIM_SAMPLE_PERIOD_MSEC-Delta;
123             if (SleepTime<0) SleepTime=0;
124             QThread::msleep(SleepTime);
125         }
126         emit SimDone();
127         emit finished();
128         qInfo()<<"After Finish";
129     }
