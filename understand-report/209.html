1     /* Dsp.h - small DSP primitives used by the timegrapher core.
2      *
3      * Two stages:
4      *   (1) tg_hpf_t      - single-pole DC blocker (rejects rumble below f_c)
5      *   (2) tg_envelope_t - full-wave rectifier + one-pole LPF for envelope
6      *
7      * The pipeline is: PCM -> hpf -> envelope -> detector.
8      * We deliberately do NOT bandpass-filter the audio. A 4th-order bandpass
9      * rings for ~3-5 ms and smears the distinct A and C events into a single
10      * blob in the envelope. Watch ticks are broadband impulses; removing DC
11      * is all the filtering we need.
12      */
13     #ifndef TG_DSP_H
14     #define TG_DSP_H
15     
16     #include <stddef.h>
17     
18     /* Single-pole highpass / DC blocker.
19      *   y[n] = x[n] - x[n-1] + a * y[n-1],   a = exp(-2*pi*f_c/fs)
20      * Cutoff f_c is typically 100-300 Hz: enough to reject AC hum and rumble,
21      * low enough that no tick energy is affected. */
22     typedef struct {
23         double a;       /* pole coefficient                          */
24         double x_prev;  /* last input sample                         */
25         double y_prev;  /* last output sample                        */
26     } tg_hpf_t;
27     
28     void tg_hpf_init   (tg_hpf_t *h, double fs, double f_c);
29     void tg_hpf_reset  (tg_hpf_t *h);
30     void tg_hpf_process(tg_hpf_t *h, const float *in, float *out, size_t n);
31     
32     /* Envelope detector: full-wave rectify + one-pole LPF.
33      *   y[n] = y[n-1] + alpha * (|x[n]| - y[n-1])
34      * Smoothing time constant controls transient sharpness. 0.15 ms is short
35      * enough to preserve A and C as distinct envelope peaks; longer values
36      * blur them together. */
37     typedef struct {
38         double alpha;   /* LPF coefficient                           */
39         double state;   /* LPF state                                 */
40     } tg_envelope_t;
41     
42     void tg_envelope_init   (tg_envelope_t *e, double fs, double smoothing_ms);
43     void tg_envelope_reset  (tg_envelope_t *e);
44     void tg_envelope_process(tg_envelope_t *e, const float *in, float *out, size_t n);
45     
46     #endif
