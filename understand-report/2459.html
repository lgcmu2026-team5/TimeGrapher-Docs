52     void TPlaybackWorker::StartPlayback(const QString &FileName)
53     {
54         int BytesIn;
55         double CurrentTime;
56         qint64 Start,Delta,SleepTime;
57     
58         if (!mTimerStarted)
59         {
60             mTimerStarted=true;
61             mTimer.start();
62         }
63     
64         QFile *file = new QFile(FileName);
65         TWaveHeader header;
66         if (!file->exists()) {
67             delete file;
68             emit PlaybackDoneReadingFile();
69             emit finished();
70             return;
71         }
72     
73         if (!file->open(QIODevice::ReadOnly))
74         {
75             delete file;
76             emit PlaybackDoneReadingFile();
77             emit finished();
78             return;
79         }
80     
81         QDataStream in(file);
82         in.setByteOrder(QDataStream::LittleEndian); // WAV is Little Endian
83     
84         file->read(header.riffId, 4);
85         in >> header.fileSize;
86         file->read(header.waveId, 4);
87         file->read(header.fmtId, 4);
88         in >> header.fmtSize;
89         in >> header.audioFormat;
90         in >> header.numChannels;
91         in >> header.sampleRate;
92         in >> header.byteRate;
93         in >> header.blockAlign;
94         in >> header.bitsPerSample;
95     
96         // Skip any extra fmt bytes if fmtSize > 16
97         if (header.fmtSize > 16) file->seek(file->pos() + (header.fmtSize - 16));
98     
99         // Look for "data" chunk (it might not be immediately after fmt)
100         char chunkId[4];
101         while (!file->atEnd()) {
102             file->read(chunkId, 4);
103             uint32_t chunkSize;
104             in >> chunkSize;
105             if (qstrncmp(chunkId, "data", 4) == 0) {
106                 header.dataSize = chunkSize;
107                 break;
108             }
109             file->seek(file->pos() + chunkSize);
110         }
111         if (qstrncmp(header.riffId, "RIFF", 4) != 0 || (header.sampleRate!=mSamplesPerSecond) ||
112             (header.numChannels!=1)|| (header.bitsPerSample != 32)||
113             (header.audioFormat != 3))
114         {
115          emit PlaybackDoneReadingFile();
116             file->close();
117             delete file;
118             emit PlaybackDoneReadingFile();
119             emit finished();
120             return;
121         }
122         int numSamples = header.dataSize / sizeof(float);
123     
124         while (!in.atEnd() && (numSamples>0))
125         {
126             Start=mTimer.elapsed();
127     
128             BytesIn=in.readRawData(mDataIn, mDataInSize);
129             if (BytesIn<0)
130             {
131                qInfo() << "Read Error ="<<BytesIn;
132                break;
133             }
134             else if ((BytesIn%4)!=0)
135             {
136               qInfo() << "Read Error not Modulus of 4";
137               break;
138             }
139             else if (BytesIn==0)
140             {
141               qInfo() << "Read Error 0";
142               break;
143             }
144             else if (QThread::currentThread()->isInterruptionRequested())
145             {
146                 break; // Exit loop early
147             }
148             unsigned int NumberOfSamples=BytesIn/SAMPLE_SIZE;
149     
150             mRawAudio->Mutex.lock();
151             unsigned int TempWriteIndex = mRawAudio->WriteIndex;
152             mRawAudio->Mutex.unlock();
153             int SamplesLeft=std::min(NumberOfSamples,mRawAudio->NumberOfAudioSamples-TempWriteIndex);
154             memcpy(&mRawAudio->Samples[TempWriteIndex], mDataIn, SamplesLeft * SAMPLE_SIZE);
155             if(SamplesLeft < NumberOfSamples)
156             {
157                 memcpy(mRawAudio->Samples, &mDataIn[SamplesLeft], (NumberOfSamples - SamplesLeft) * SAMPLE_SIZE);
158                 qInfo() << "MasterPlaybackData Samples Rollover";
159             }
160             mRawAudio->Mutex.lock();
161             mRawAudio->WriteIndex = (TempWriteIndex+ NumberOfSamples) %  mRawAudio->NumberOfAudioSamples;
162             mRawAudio->TotalSamplesWritten+=NumberOfSamples;
163             mRawAudio->Mutex.unlock();
164             emit PlaybackDataReady(); // Emit data to the main thread
165     
166             ++mFrameCount;
167             mSampleCount+=NumberOfSamples;
168             numSamples-=NumberOfSamples;
169             CurrentTime = mTimer.elapsed()/1000.0;
170             if (CurrentTime-mLastTime > 2) // average fps over 2 seconds
171             {
172                 double fdelta;
173                 fdelta=CurrentTime-mLastTime;
174                 mRawAudio->FPS=mFrameCount/fdelta;
175                 mRawAudio->SPS=mSampleCount/fdelta;
176                 mRawAudio->SPF=mSampleCount/mFrameCount;
177                 mLastTime=CurrentTime;
178                 mFrameCount=0;
179                 mSampleCount=0;
180             }
181             Delta=(mTimer.elapsed()-Start)+DELAY_FUGE_TIME_MS;
182             SleepTime=PLAYBACK_SAMPLE_PERIOD_MSEC-Delta;
183             if (SleepTime<0) SleepTime=0;
184             QThread::msleep(SleepTime);
185         }
186         qInfo()<<"Before Close";
187         file->close();
188         delete file;
189         emit PlaybackDoneReadingFile();
190         emit finished();
191         qInfo()<<"After Finish";
192     }
