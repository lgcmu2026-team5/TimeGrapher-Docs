1     /* Timegrapher.cpp -- Public API entry point and pipeline glue.
2      *
3      * This translation unit owns the public tg_* API and the orchestration
4      * that wires the per-stage primitives (Dsp, Detector, Bph) into a
5      * single streaming pipeline. The actual per-stage work happens in
6      * the corresponding .cpp files.
7      *
8      * Pipeline (per call to tg_process):
9      *
10      *   1. DC blocker          -- single-pole HPF (Dsp.cpp), default 200 Hz.
11      *                             Removes mic wind, table rumble, mains hum.
12      *
13      *   2. Envelope            -- full-wave rectify + 0.15 ms one-pole LPF
14      *                             (Dsp.cpp). Output is the slow-moving
15      *                             envelope of the impulse activity.
16      *
17      *   3. Detector            -- silence-based burst detector (Detector.cpp).
18      *                             Emits A and C events with sub-sample
19      *                             timing, plus V5.4 C-onset metadata.
20      *
21      *   4. BPH detection       -- Rayleigh phase-score over event history
22      *                             (Bph.cpp). Once a candidate exceeds the
23      *                             lock threshold AND clears the V5.3 median
24      *                             A-to-A guard, sync state moves to SYNCED.
25      *
26      *   5. Sync PLL            -- once locked, a small period-tracking PLL
27      *                             refines the beat period to follow drift.
28      *                             Sync is lost after a configurable number
29      *                             of consecutive missed expected events.
30      *
31      *   6. Delay-line / public events -- tg_event_t emitted to the caller.
32      *                             A delay line is used to align reported
33      *                             event times to the input PCM stream
34      *                             (envelope LPF introduces a small group
35      *                             delay; we compensate so the timestamps
36      *                             line up with what the user sees in their
37      *                             waveform display).
38      *
39      * The library only emits timing events here. Per-event amplitude,
40      * predicted-time pairing gates, and window-averaged readouts live in
41      * libtgan and libtgacf respectively, both built on top of this API.
42      *
43      * Thread safety: a single tg_context_t is owned by the calling thread.
44      * Multiple streams = multiple contexts, no shared global state.
45      *
46      * V5.4 additions handled here:
47      *   - tg_c_placement_t enum and the c_placement field on tg_config_t.
48      *   - tg_set_c_placement / tg_get_c_placement runtime control.
49      *   - Promotion of raw events (which always carry both peak and onset
50      *     timings on C) to public tg_event_t per the c_placement mode.
51      *     PEAK -> primary trio = peak; ONSET -> primary trio = onset (with
52      *     silent fallback to peak if onset detection failed for that beat).
53      *     Onset metadata fields are populated identically in both modes.
54      *
55      * V5.5: file renamed from timegrapher.c to Timegrapher.cpp; compiles
56      * under C++17. The public API in Timegrapher.h is wrapped in
57      * extern "C" so the ABI is identical to V5.4 for C, C++, Rust,
58      * Python (cffi), and other foreign-function callers.
59      *
60      * V5.6: regime-change reset coordination. After each detector pass
61      * we consume the detector's regime_reset_pending flag (via
62      * tg_detector_consume_regime_reset). On a true return we flush our
63      * library-level state -- event history, sync/PLL internals, cached
64      * BPH/period -- and the detector's own adaptive state (peak median,
65      * noise floor), while preserving the detector's sample clock and
66      * env_ring abs-index reference so timestamps and the C-onset
67      * backward walk stay correct. The triggering peak is stamped into
68      * the regime ring so the next beat doesn't immediately re-trigger.
69      * Surfaces as tg_result_t.detector_reset_event.
70      */

138     static int ensure_buf(tg_context_t *ctx, size_t n) {
139         if (n <= ctx->buf_capacity) return 0;
140         size_t new_cap = ctx->buf_capacity ? ctx->buf_capacity : TG_INITIAL_BUF;
141         while (new_cap < n) new_cap *= 2;
142         float *nf = (float*)realloc(ctx->buf_filt, new_cap * sizeof(float));
143         if (!nf) return -1;
144         ctx->buf_filt = nf;
145         float *ne = (float*)realloc(ctx->buf_env, new_cap * sizeof(float));
146         if (!ne) return -1;
147         ctx->buf_env = ne;
148         ctx->buf_capacity = new_cap;
149         return 0;
150     }

189     /* Implements a FIFO of fixed length D = ctx->delay_samples. Each input
190      * sample is enqueued; if the queue has more than D samples, the oldest
191      * is popped and written to `out`. */
192     static size_t delay_push_pop(tg_context_t *ctx,
193                                  const float *in, size_t n,
194                                  float *out)
195     {
196         size_t cap = ctx->delay_capacity;
197         size_t D   = ctx->delay_samples;
198         if (cap == 0 || n == 0) return 0;
199         size_t produced = 0;
200         for (size_t i = 0; i < n; ++i) {
201             if (ctx->delay_filled >= D) {
202                 size_t read_idx = (ctx->delay_write_idx + cap - D) % cap;
203                 out[produced++] = ctx->delay_buf[read_idx];
204             } else {
205                 ctx->delay_filled++;
206             }
207             ctx->delay_buf[ctx->delay_write_idx] = in[i];
208             ctx->delay_write_idx = (ctx->delay_write_idx + 1) % cap;
209             ctx->total_env_samples_in++;
210         }
211         return produced;
212     }

216     static void push_event_history(tg_context_t *ctx, double t) {
217         ctx->ev_history[ctx->ev_history_head] = t;
218         ctx->ev_history_head = (ctx->ev_history_head + 1) % TG_EVENT_HISTORY;
219         if (ctx->ev_history_count < TG_EVENT_HISTORY) ctx->ev_history_count++;
220     }

222     static void copy_history_linear(const tg_context_t *ctx, double *out) {
223         size_t cnt = ctx->ev_history_count;
224         if (cnt == 0) return;
225         if (cnt < TG_EVENT_HISTORY) {
226             memcpy(out, ctx->ev_history, cnt * sizeof(double));
227         } else {
228             size_t head = ctx->ev_history_head;
229             size_t tail_len = TG_EVENT_HISTORY - head;
230             memcpy(out,            ctx->ev_history + head, tail_len * sizeof(double));
231             memcpy(out + tail_len, ctx->ev_history,        head     * sizeof(double));
232         }
233     }

257     tg_context_t *tg_init(const tg_config_t *cfg_in) {
258         if (!cfg_in || cfg_in->sample_rate <= 0.0) return NULL;
259     
260         if (cfg_in->bph_mode == TG_BPH_MODE_MANUAL &&
261             !tg_is_valid_manual_bph(cfg_in->manual_bph)) {
262             return NULL;
263         }
264     
265         tg_context_t *ctx = (tg_context_t*)calloc(1, sizeof(*ctx));
266         if (!ctx) return NULL;
267         ctx->cfg = *cfg_in;
268     
269         /* Apply zero-defaults at runtime */
270         if (ctx->cfg.hpf_cutoff_hz           == 0.0) ctx->cfg.hpf_cutoff_hz           = 200.0;
271         if (ctx->cfg.envelope_smooth_ms      == 0.0) ctx->cfg.envelope_smooth_ms      = 0.15;
272         if (ctx->cfg.event_min_separation_ms == 0.0) ctx->cfg.event_min_separation_ms = 2.0;
273         if (ctx->cfg.sync_tolerance_pct      == 0.0) ctx->cfg.sync_tolerance_pct      = 3.0;
274         if (ctx->cfg.auto_detect_seconds     == 0.0) ctx->cfg.auto_detect_seconds     = 1.5;
275         if (ctx->cfg.sync_loss_misses        == 0)   ctx->cfg.sync_loss_misses        = 12;
276         if (ctx->cfg.pll_period_gain         == 0.0) ctx->cfg.pll_period_gain         = 0.01;
277         if (ctx->cfg.pll_ac_gain             == 0.0) ctx->cfg.pll_ac_gain             = 0.05;
278     
279         tg_hpf_init     (&ctx->hpf, ctx->cfg.sample_rate, ctx->cfg.hpf_cutoff_hz);
280         tg_envelope_init(&ctx->env, ctx->cfg.sample_rate, ctx->cfg.envelope_smooth_ms);
281         tg_detector_init(&ctx->det, ctx->cfg.sample_rate);
282     
283         /* Init-time threshold fractions */
284         if (ctx->cfg.onset_fraction_init    > 0.0)
285             tg_detector_set_onset_fraction   (&ctx->det, ctx->cfg.onset_fraction_init);
286         if (ctx->cfg.min_peak_fraction_init > 0.0)
287             tg_detector_set_min_peak_fraction(&ctx->det, ctx->cfg.min_peak_fraction_init);
288     
289         tg_sync_init(&ctx->sync);
290     
291         /* Envelope delay line: 50 ms, sample-rate dependent. */
292         ctx->delay_capacity = (size_t)(0.050 * ctx->cfg.sample_rate);
293         if (ctx->delay_capacity < 16) ctx->delay_capacity = 16;
294         ctx->delay_samples = ctx->delay_capacity;
295         ctx->delay_buf = (float*)calloc(ctx->delay_capacity, sizeof(float));
296         if (!ctx->delay_buf) { tg_destroy(ctx); return NULL; }
297     
298         if (ensure_buf(ctx, TG_INITIAL_BUF) || ensure_raw_events(ctx, 64) ||
299             ensure_out_events(ctx, 64))
300         {
301             tg_destroy(ctx);
302             return NULL;
303         }
304         return ctx;
305     }

319     int tg_process(tg_context_t *ctx,
320                    const float  *pcm, size_t num_samples,
321                    tg_result_t  *result)
322     {
323         if (!ctx || !result) return -1;
324         memset(result, 0, sizeof(*result));
325     
326         /* DSP -> envelope -> delay line.
327          *
328          * Detector reads the un-delayed envelope (so events have absolute
329          * timing in the caller's input stream). The 50 ms delay line is
330          * applied separately to produce the user-facing processed_pcm,
331          * which lags events by ~50 ms but is aligned for plotting (each
332          * event's sample_index, when subtracted from processed_pcm_start_sample,
333          * gives the offset into the buffer where the impulse will appear).
334          */
335         size_t env_out_len = 0;
336         if (num_samples > 0) {
337             if (ensure_buf(ctx, num_samples) != 0) return -2;
338             if (ensure_env_out(ctx, num_samples) != 0) return -2;
339             tg_hpf_process     (&ctx->hpf, pcm,            ctx->buf_filt, num_samples);
340             tg_envelope_process(&ctx->env, ctx->buf_filt,  ctx->buf_env,  num_samples);
341             env_out_len = delay_push_pop(ctx, ctx->buf_env, num_samples,
342                                          ctx->buf_env_out);
343         }
344     
345         /* Compute absolute index of first sample of the delayed envelope */
346         uint64_t input_end   = ctx->total_env_samples_in;
347         uint64_t output_end  = (input_end > ctx->delay_samples)
348                              ? (input_end - ctx->delay_samples) : 0;
349         uint64_t output_start = output_end - env_out_len;
350     
351         result->processed_pcm              = ctx->buf_env_out;
352         result->processed_pcm_len          = env_out_len;
353         result->processed_pcm_start_sample = output_start;
354     
355         /* Detector reads the un-delayed envelope. Events emitted have
356          * timing in the caller's absolute stream (in seconds since the
357          * first input sample given to tg_init). */
358         size_t raw_count = 0;
359         if (num_samples > 0) {
360             size_t window = (size_t)ctx->det.min_silence_samples;
361             if (window < 16) window = 16;
362             size_t worst = (2 * num_samples) / window + 4;
363             if (ensure_raw_events(ctx, worst) != 0) return -2;
364             size_t got = 0;
365             tg_detector_process(&ctx->det,
366                                 ctx->buf_env, num_samples,
367                                 ctx->raw_events,
368                                 &got, ctx->raw_events_capacity);
369             raw_count = got;
370         }
371     
372         /* V5.6: regime-change reset.
373          *
374          * The detector flags a regime change when a new burst peak is
375          * >= 10x the rolling min of recent peaks (typical case: quiet
376          * ambient prelude -> watch arriving on microphone). When that
377          * happens the adaptive state (event history, sync, noise floor,
378          * peak median ring) is polluted with values from the previous
379          * regime and would take many seconds to recover.
380          *
381          * The fix: on a flagged change, flush all polluted adaptive
382          * state, then surface a one-shot detector_reset_event so the
383          * caller can update UI. The raw events emitted this batch are
384          * kept (the triggering event is genuine -- the detector found
385          * something loud enough to fire), but the BPH/sync state starts
386          * fresh from this event onward.
387          *
388          * Note: tg_detector_reset internally re-seeds the regime ring
389          * with the triggering peak so the next beat won't re-trigger. */
390         if (tg_detector_consume_regime_reset(&ctx->det)) {
391             result->detector_reset_event = 1;
392     
393             /* Capture the triggering peak (the most recent value pushed
394              * into the regime ring) before flushing the detector, so we
395              * can re-seed the ring after the reset. */
396             double seed_peak = ctx->det.burst_max;
397     
398             /* Flush detector adaptive state. This clears peak_history,
399              * noise_history, sync-internal state in the detector itself.
400              * The env_ring is also cleared but stays allocated.
401              *
402              * IMPORTANT: tg_detector_reset zeros several pieces of
403              * streaming state that are correct to zero at stream start
404              * but wrong when called mid-stream from here:
405              *
406              *   - total_samples: the detector's sample clock. If reset,
407              *     subsequent events would get stamped with absolute sample
408              *     indices starting from 0, breaking timestamp continuity.
409              *
410              *   - env_ring_newest_abs / env_ring_has_data: the absolute
411              *     sample-index reference for the C-onset backward-walk
412              *     ring. If reset, the backward walk would compute wrong
413              *     ring offsets (or skip the walk entirely until refilled).
414              *
415              * Save and restore these around the reset. */
416             uint64_t saved_total = ctx->det.total_samples;
417             uint64_t saved_env_newest = ctx->det.env_ring_newest_abs;
418             int      saved_env_has   = ctx->det.env_ring_has_data;
419             tg_detector_reset(&ctx->det);
420             ctx->det.total_samples       = saved_total;
421             ctx->det.env_ring_newest_abs = saved_env_newest;
422             ctx->det.env_ring_has_data   = saved_env_has;
423     
424             /* Re-seed the regime ring so the next beat doesn't re-trip
425              * the comparator. Without this, the first post-reset beat
426              * would have an empty ring (min would be 0), and the second
427              * beat would trip again. */
428             ctx->det.regime_peak_ring[0] = seed_peak;
429             ctx->det.regime_peak_count   = 1;
430             ctx->det.regime_peak_head    = 1;
431     
432             /* Flush library-level state too: event history, sync state,
433              * cached BPH and beat period. Use tg_sync_reset to fully
434              * clear the PLL internals (next_a_time, period_est, etc.),
435              * not just the synced flag -- otherwise the post-reset
436              * watchdog uses stale predictions and immediately unsyncs. */
437             ctx->current_bph         = 0;
438             ctx->current_beat_period = 0.0;
439             ctx->current_ac_offset   = 0.0;
440             ctx->ev_history_count    = 0;
441             ctx->ev_history_head     = 0;
442             tg_sync_reset(&ctx->sync);
443     
444             /* Suppress the raw events from THIS batch. They were emitted
445              * on stale thresholds just before the regime change was
446              * detected; better to start fresh from the next batch. */
447             raw_count = 0;
448         }
449     
450         /* Push A events into BPH history */
451         for (size_t i = 0; i < raw_count; ++i) {
452             if (ctx->raw_events[i].is_onset)
453                 push_event_history(ctx, ctx->raw_events[i].time_seconds);
454         }
455     
456         /* BPH detection: if not already synced, see if we have enough history. */
457         int try_detect = 0;
458         if (!ctx->sync.synced && ctx->ev_history_count >= 8) {
459             double tmp[TG_EVENT_HISTORY];
460             copy_history_linear(ctx, tmp);
461             double earliest = tmp[0];
462             double latest   = tmp[ctx->ev_history_count - 1];
463             if (latest - earliest >= ctx->cfg.auto_detect_seconds) try_detect = 1;
464         }
465     
466         if (!ctx->sync.synced && try_detect) {
467             double tmp[TG_EVENT_HISTORY];
468             copy_history_linear(ctx, tmp);
469             int    matched = 0;
470             double phase_score = 0.0;
471             double matched_period = 0.0;
472             if (ctx->cfg.bph_mode == TG_BPH_MODE_AUTO) {
473                 matched = tg_bph_pick_by_phase(tmp, ctx->ev_history_count,
474                                                TG_AUTO_BPH_LIST, TG_AUTO_BPH_COUNT,
475                                                0.7, &phase_score, &matched_period);
476             } else {
477                 int u = ctx->cfg.manual_bph;
478                 double T = 3600.0 / (double)u;
479                 phase_score = tg_phase_score(tmp, ctx->ev_history_count, T);
480                 if (phase_score >= 0.7) {
481                     matched = u; matched_period = T;
482                 }
483             }
484             if (matched) {
485                 double half = 0.5 * matched_period;
486                 double sum_small = 0.0; size_t cnt_small = 0;
487                 for (size_t i = 1; i < ctx->ev_history_count; ++i) {
488                     double d = tmp[i] - tmp[i-1];
489                     if (d > 0.0 && d < half) { sum_small += d; cnt_small++; }
490                 }
491                 double ac = (cnt_small > 0) ? sum_small / (double)cnt_small
492                                             : matched_period * 0.05;
493                 double tol = matched_period * ctx->cfg.sync_tolerance_pct * 0.01;
494                 double last_ev = tmp[ctx->ev_history_count - 1];
495                 tg_sync_lock(&ctx->sync, matched, matched_period, ac,
496                              last_ev, tol, ctx->cfg.sync_loss_misses,
497                              ctx->cfg.pll_period_gain, ctx->cfg.pll_ac_gain);
498                 ctx->current_bph         = matched;
499                 ctx->current_beat_period = matched_period;
500                 ctx->current_ac_offset   = ac;
501                 result->sync_acquired_event = 1;
502     
503                 /* Tighten the silence and A-to-A gates now that BPH is known. */
504                 tg_detector_set_min_silence(&ctx->det, 0.4 * matched_period);
505                 tg_detector_set_min_a_interval(&ctx->det, 0.7 * matched_period);
506     
507                 /* V5.2: tune C-search skip from beat period. The default
508                  * 3 ms handles the common "first sub-impulse" case. Once
509                  * BPH is known, scale to ~3% of beat period: long enough
510                  * to skip past common intermediate sub-impulses (which
511                  * appear within the first ~5 ms of a burst), short enough
512                  * to stay safely below the C peak at t_AC even for
513                  * knocking-amplitude watches.
514                  *
515                  * Approximate skip values (3% of period):
516                  *   18000 BPH (period 200 ms) -> 6.0 ms
517                  *   21600 BPH (period 167 ms) -> 5.0 ms
518                  *   28800 BPH (period 125 ms) -> 3.8 ms
519                  *   36000 BPH (period 100 ms) -> 3.0 ms
520                  *
521                  * For comparison, t_AC at 360 deg amplitude (knocking):
522                  *   18000 BPH lift 50 -> 9.2 ms
523                  *   21600 BPH lift 50 -> 7.4 ms
524                  *   28800 BPH lift 52 -> 5.7 ms */
525                 double skip_s = 0.03 * matched_period;
526                 tg_detector_set_c_search_skip(&ctx->det, skip_s);
527             }
528         }
529     
530         /* Run sync tracker and detect loss */
531         int prev_synced = ctx->sync.synced;
532         if (ctx->sync.synced) {
533             for (size_t i = 0; i < raw_count; ++i) {
534                 tg_sync_update(&ctx->sync, ctx->raw_events[i].time_seconds);
535                 if (!ctx->sync.synced) break;
536             }
537         }
538         /* Time-based sync loss watchdog */
539         if (ctx->sync.synced) {
540             double stream_t = (double)(input_end) / ctx->cfg.sample_rate;
541             double tol = ctx->current_beat_period * ctx->cfg.sync_tolerance_pct * 0.01;
542             if (stream_t > ctx->sync.next_a_time
543                 + ctx->cfg.sync_loss_misses * ctx->current_beat_period + tol)
544             {
545                 ctx->sync.synced = 0;
546             }
547         }
548         if (prev_synced && !ctx->sync.synced) {
549             result->sync_lost_event = 1;
550             ctx->current_bph = 0;
551             ctx->current_beat_period = 0.0;
552             ctx->current_ac_offset = 0.0;
553             ctx->ev_history_count = 0;
554             ctx->ev_history_head  = 0;
555             tg_detector_set_min_silence  (&ctx->det, 0.020);
556             tg_detector_set_min_a_interval(&ctx->det, 0.0);
557             tg_detector_set_c_search_skip(&ctx->det, 0.003);  /* back to default */
558         }
559     
560         /* Build sync_status */
561         if (ctx->cfg.bph_mode == TG_BPH_MODE_MANUAL) {
562             if (ctx->sync.synced) {
563                 result->sync_status  = TG_SYNC_SYNCED;
564                 result->detected_bph = ctx->cfg.manual_bph;
565             } else if (try_detect) {
566                 result->sync_status  = TG_SYNC_MISMATCH;
567                 result->detected_bph = ctx->cfg.manual_bph;
568             } else {
569                 result->sync_status  = TG_SYNC_NOT_SYNCED;
570                 result->detected_bph = ctx->cfg.manual_bph;
571             }
572         } else {
573             if (ctx->sync.synced) {
574                 result->sync_status  = TG_SYNC_SYNCED;
575                 result->detected_bph = ctx->current_bph;
576             } else {
577                 result->sync_status  = TG_SYNC_NOT_SYNCED;
578                 result->detected_bph = 0;
579             }
580         }
581         result->measured_period_s = ctx->current_beat_period;
582     
583         /* Emit events: copy timing, set is_pre_sync, optionally drop. */
584         size_t emit_count = 0;
585         if (raw_count > 0) {
586             if (ensure_out_events(ctx, raw_count) != 0) return -2;
587             int pre_sync = (ctx->current_bph <= 0);
588             for (size_t i = 0; i < raw_count; ++i) {
589                 if (pre_sync && ctx->cfg.suppress_pre_sync_events) continue;
590                 tg_event_t *ev = &ctx->out_events[emit_count++];
591                 const tg_raw_event_t *r = &ctx->raw_events[i];
592                 int is_a = r->is_onset;
593     
594                 ev->type        = is_a ? TG_EVENT_A : TG_EVENT_C;
595                 ev->is_pre_sync = pre_sync;
596                 ev->peak_value  = r->peak_value;
597     
598                 /* V5.4: choose primary timing per c_placement (C only).
599                  * For A events the raw timing IS the onset; nothing to
600                  * choose. For C events:
601                  *   PEAK  -> primary = peak (raw event's main fields)
602                  *   ONSET -> primary = onset, falling back to peak if
603                  *            onset detection failed for this beat */
604                 int use_onset_primary =
605                     (!is_a)
606                     && (ctx->cfg.c_placement == TG_C_PLACEMENT_ONSET)
607                     && (r->onset_valid != 0);
608     
609                 if (use_onset_primary) {
610                     ev->sample_index      = r->onset_sample_index;
611                     ev->sub_sample_offset = r->onset_sub_sample_offset;
612                     ev->time_seconds      = r->onset_time_seconds;
613                 } else {
614                     ev->sample_index      = r->sample_index;
615                     ev->sub_sample_offset = r->sub_sample_offset;
616                     ev->time_seconds      = r->time_seconds;
617                 }
618     
619                 /* V5.4: always populate onset metadata fields. For A
620                  * events these are zeroed at the detector. For C events
621                  * they reflect the onset detection result. */
622                 ev->onset_sample_index      = r->onset_sample_index;
623                 ev->onset_sub_sample_offset = r->onset_sub_sample_offset;
624                 ev->onset_time_seconds      = r->onset_time_seconds;
625                 ev->onset_valid             = r->onset_valid;
626             }
627         }
628         result->events     = ctx->out_events;
629         result->num_events = emit_count;
630     
631         /* Detector state for diagnostics */
632         {
633             double onset_thr, min_peak_thr, eff_noise, ref_peak;
634             tg_detector_get_thresholds(&ctx->det, &onset_thr, &min_peak_thr,
635                                        &eff_noise, &ref_peak);
636             result->onset_threshold    = (float)onset_thr;
637             result->min_peak_threshold = (float)min_peak_thr;
638             result->noise_floor        = (float)eff_noise;
639             result->reference_peak     = (float)ref_peak;
640         }
641     
642         ctx->total_samples_processed += num_samples;
643         return 0;
644     }

646     int tg_flush(tg_context_t *ctx, tg_result_t *result) {
647         if (!ctx || !result) return -1;
648         size_t n = ctx->delay_samples;
649         if (ctx->det.burst_end_samples > n) n = ctx->det.burst_end_samples;
650         n += 32;
651         if (n == 0) { memset(result, 0, sizeof(*result)); return 0; }
652         float *silent = (float*)calloc(n, sizeof(float));
653         if (!silent) return -2;
654         int rc = tg_process(ctx, silent, n, result);
655         free(silent);
656         return rc;
657     }

659     void tg_reset(tg_context_t *ctx) {
660         if (!ctx) return;
661         tg_hpf_reset     (&ctx->hpf);
662         tg_envelope_reset(&ctx->env);
663         tg_detector_reset(&ctx->det);
664         tg_detector_set_min_silence  (&ctx->det, 0.020);
665         tg_detector_set_min_a_interval(&ctx->det, 0.0);
666         tg_sync_init(&ctx->sync);
667         ctx->current_bph         = 0;
668         ctx->current_beat_period = 0.0;
669         ctx->current_ac_offset   = 0.0;
670         ctx->ev_history_count    = 0;
671         ctx->ev_history_head     = 0;
672         if (ctx->delay_buf && ctx->delay_capacity)
673             memset(ctx->delay_buf, 0, ctx->delay_capacity * sizeof(float));
674         ctx->delay_write_idx       = 0;
675         ctx->delay_filled          = 0;
676         ctx->total_env_samples_in  = 0;
677         ctx->total_samples_processed = 0;
678     }
