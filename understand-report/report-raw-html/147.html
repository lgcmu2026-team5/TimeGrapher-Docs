1     /* Detector.h - silence-based onset detector.
2      *
3      * Segments the envelope into SILENCE and BURST regions and emits two
4      * events per accepted burst:
5      *
6      *   A (is_onset = 1) : silence -> signal threshold CROSSING.
7      *       Sub-sample timing by linear interpolation between the two
8      *       samples straddling the onset threshold. This is the moment
9      *       the envelope first lifts above the ambient noise floor and
10      *       corresponds to the escapement unlock.
11      *
12      *   C (is_onset = 0) : PEAK of the burst envelope.
13      *       Sub-sample timing by parabolic interpolation across the max
14      *       sample and its two neighbors. Corresponds to the escape
15      *       wheel dropping onto the locking face (the loudest noise).
16      *
17      * Thresholds are scale-invariant and derived from two robust estimators:
18      *
19      *   noise_floor    = 75th percentile of downsampled silence-region
20      *                    envelope samples (a 256-slot ring buffer sampled
21      *                    once per millisecond during silence). Tracks the
22      *                    TYPICAL ambient level between ticks, not its
23      *                    minimum -- crucial for preventing false onsets in
24      *                    quiet sections where rumble hovers above the min.
25      *
26      *   reference_peak = median of the last 16 accepted burst peaks.
27      *                    Robust to outliers (one loud bang doesn't move it)
28      *                    and to sparse ticks (no decay during long gaps).
29      *
30      * No per-recording tuning is needed: the same gate fractions (3% onset,
31      * 2% release, 20% min-peak) are used for all watches.
32      *
33      * Both the silence gate (time between beats) and the burst-end gate
34      * (time within a beat) are WALL-CLOCK based -- counted in samples
35      * since a reference point, not continuous-below-threshold samples.
36      * This handles watches whose ring-down oscillates near the release
37      * threshold without settling.
38      *
39      * A is held internally until burst-end so that bursts whose peak fails
40      * the min-peak threshold (noise bumps) can be discarded entirely,
41      * without leaving an orphan A in the output.
42      */

122         /* ---- V5.6: regime-change detector --------------------------------
123          *
124          * Detects a >= 10x jump in burst peak amplitude relative to the
125          * rolling minimum of the last few accepted peaks. Trips when audio
126          * crosses from one acoustic regime to another (e.g. quiet
127          * ambient noise -> watch on microphone, or removed and replaced).
128          *
129          * On trip, the library flushes the polluted adaptive state
130          * (event history, sync state, noise floor ring, peak median ring)
131          * and re-seeds with the triggering peak so the next beat doesn't
132          * immediately re-trigger.
133          *
134          * regime_peak_ring is independent of peak_history -- peak_history
135          * gets flushed on reset, but regime_peak_ring keeps the seeded
136          * trigger value so the next-beat comparison has a baseline.
137          *
138          * Tunables:
139          *   - ratio threshold: 10x (TG_REGIME_RATIO)
140          *   - ring size: 8 entries (TG_REGIME_RING_N) -- ~1.5 s at 21600 BPH
141          *   - cooldown: 1 second between resets, gated by sample counter
142          *   - absolute floor: 0.001 -- below this, both peaks are noise
143          *     and the ratio check is skipped (TG_REGIME_FLOOR) */
144         double   regime_peak_ring[TG_REGIME_RING_N];
145         int      regime_peak_count;
146         int      regime_peak_head;
147         uint64_t regime_last_reset_idx;  /* abs sample idx of last reset, 0 = never */
148         int      regime_reset_pending;    /* set on trip, cleared by lib after flush */

168         int      have_peak_plus1;

170         /* V5.2: separate C-peak tracking. The "burst_max" above tracks the
171          * absolute envelope max within a burst; that's used for the min-peak
172          * height check at burst-end. But for C-event TIMING we need to skip
173          * the first N samples of the burst, which often contain the A's own
174          * impulse spike or intermediate sub-impulses that are NOT the C
175          * cluster (especially on multi-impulse watches where one of these
176          * earlier peaks can occasionally exceed the height of the actual C).
177          *
178          * burst_c_idx / burst_c_max track the envelope max ONLY in samples
179          * past `c_search_skip_samples` from burst_start. If no peak is
180          * found in that region (c_have_peak == 0), C falls back to
181          * burst_max_idx (preserving behavior for very short bursts and
182          * pre-sync recordings where the skip wasn't tuned).
183          *
184          * Default skip is 3 ms (covers the common "first sub-impulse
185          * spike" case but stays below physical t_AC_min for any
186          * reasonable amplitude). The library sets a longer, BPH-tuned
187          * skip after sync (~80% of t_AC_min for max plausible amplitude). */
188         uint64_t c_search_skip_samples;
189         int      c_have_peak;
190         double   burst_c_max;
191         uint64_t burst_c_idx;
192         double   burst_c_y_minus1;
193         double   burst_c_y_plus1;
194         int      c_have_peak_plus1;

196         /* V5.4: C-onset detection via backward walk from burst_c_idx.
197          *
198          * env_ring is a small ring buffer of recent envelope samples used
199          * to walk backward from the C peak at burst-emission time. We
200          * walk back looking for the most recent sample where the envelope
201          * crossed (rising) the half-of-peak threshold. Safeguards:
202          *   - "Avoid small gaps": require min_dwell_samples consecutive
203          *     samples below threshold before declaring an onset boundary
204          *     (otherwise within-cluster ringing notches register as onsets).
205          *   - "Don't go back too far": bound the walk to
206          *     c_onset_search_max_samples. Beyond that we risk crossing
207          *     into A territory.
208          *
209          * env_ring_capacity is chosen at init to cover the worst-case
210          * search distance plus the burst duration that would precede
211          * burst_c_idx in time (10 ms is plenty: at 48 kHz that's 480
212          * samples; at 384 kHz it's 3840). */
213         float   *env_ring;
214         size_t   env_ring_capacity;
215         size_t   env_ring_head;          /* next write position; wraps */
216         uint64_t env_ring_newest_abs;    /* abs sample index of most-recent write; 0 if empty */
217         int      env_ring_has_data;      /* 1 once at least one sample has been written */
218     
219         /* C-onset detection parameters. */
220         uint64_t c_onset_dwell_samples;        /* default ~0.3 ms */
221         uint64_t c_onset_search_max_samples;   /* default 5 ms pre-sync, ~t_AC/2 post-sync */

223         /* misc */
224         double   prev_sample;
225         uint64_t last_event_sample;
226         uint64_t total_samples;

315     /* V5.6: regime-change reset coordination.
316      *
317      * The detector internally tracks burst peak history (separate from
318      * the adaptive median ring). When a new peak arrives that's >=
319      * TG_REGIME_RATIO (10x) times the rolling minimum of recent peaks
320      * AND both are above TG_REGIME_FLOOR, the regime_reset_pending flag
321      * is set. This is the detector's signal to the library that
322      * "acoustic regime has changed -- the adaptive state is now stale."
323      *
324      * The library queries this via tg_detector_consume_regime_reset,
325      * which atomically reads-and-clears the flag. On a true return,
326      * the library flushes its higher-level state (event history,
327      * sync/PLL state) and calls tg_detector_reset on the detector to
328      * flush the adaptive thresholds.
329      *
330      * Returns 1 if a reset was pending (and clears the flag), 0 otherwise. */
331     int tg_detector_consume_regime_reset(tg_detector_t *d);
