1     /* Bph.cpp -- BPH (beats-per-hour) detection and sync tracking.
2      *
3      * This translation unit owns three concerns:
4      *
5      *   1. The candidate BPH lists exposed via the public API
6      *      (TG_AUTO_BPH_LIST and TG_MANUAL_BPH_LIST) and helpers used by
7      *      the host application to populate UI dropdowns.
8      *
9      *   2. The "phase score" routine (tg_phase_score) used during BPH
10      *      auto-detection. For each candidate beat period T and the
11      *      recent A-event history, we compute the magnitude of the
12      *      Rayleigh circular mean of (event_time mod T) -- the strength
13      *      of phase alignment to a hypothetical period-T grid. The
14      *      candidate with the highest score above the lock threshold wins.
15      *
16      *      V5.3 fix: this returns R1 (first harmonic) only. Earlier
17      *      versions returned R1 + R2 to gain robustness on noisy signals
18      *      that emitted both A and C events. After the V5.0 split that
19      *      restricted the BPH path to A-only events, R2 became actively
20      *      harmful: for a perfectly periodic A-only stream, R2 doubles
21      *      the score for half-period candidates (all events project to
22      *      phase 0 in the doubled-frequency space). On 18000 BPH watches
23      *      with intermittent secondary echoes, this caused mis-detection
24      *      as 36000 BPH. See docs/design.md section 4 for the full
25      *      derivation.
26      *
27      *   3. tg_bph_pick_by_phase: orchestrates the candidate sweep and
28      *      applies the V5.3 median-A-to-A-interval guard, rejecting any
29      *      candidate period less than 0.7 * median(A-to-A) as physically
30      *      implausible. The median is a robust lower bound on the true
31      *      beat period (spurious extra events can only make consecutive
32      *      A-to-A gaps smaller, never larger). Even where R1 still has
33      *      half-period aliasing modes for perfectly periodic events, the
34      *      median guard breaks the tie in favor of the true period.
35      *
36      * The sync state machine itself (acquire / track / lose) lives in
37      * Timegrapher.cpp -- this file just supplies the period-detection
38      * primitive it calls per block.
39      *
40      * V5.5: file renamed from bph.c to Bph.cpp; compiles under C++17.
41      * No algorithmic change. The public symbols (TG_AUTO_BPH_LIST,
42      * TG_MANUAL_BPH_LIST, tg_phase_score, tg_bph_pick_by_phase) keep
43      * their C-style ABI via extern "C" guards in the headers.
44      */

82     int tg_bph_match(double candidate_bph,
83                      const int *list, size_t list_len,
84                      double tolerance_pct)
85     {
86         if (candidate_bph <= 0.0 || list_len == 0) return 0;
87     
88         int best = 0;
89         double best_err = 1e30;
90         for (size_t i = 0; i < list_len; ++i) {
91             double err = fabs(candidate_bph - (double)list[i]) / (double)list[i];
92             if (err < best_err) {
93                 best_err = err;
94                 best = list[i];
95             }
96         }
97         if (best_err * 100.0 <= tolerance_pct) return best;
98         return 0;
99     }

103     static int dcmp(const void *a, const void *b) {
104         double x = *(const double*)a, y = *(const double*)b;
105         return (x < y) ? -1 : (x > y) ? 1 : 0;
106     }

108     double tg_estimate_beat_period(const double *event_times, size_t n) {
109         if (n < 4) return 0.0;
110         /* For events ACACAC..., t[i+2] - t[i] == beat period. */
111         size_t m = n - 2;
112         double *deltas = (double*)malloc(m * sizeof(double));
113         if (!deltas) return 0.0;
114         for (size_t i = 0; i < m; ++i) {
115             deltas[i] = event_times[i + 2] - event_times[i];
116         }
117         qsort(deltas, m, sizeof(double), dcmp);
118         double med = deltas[m / 2];
119         free(deltas);
120         return med;
121     }

123     /* Phase score: fold event times mod period and measure how concentrated
124      * the resulting phases are using the standard Rayleigh statistic.
125      *
126      *   ph_i  = 2*pi * (t_i mod T) / T
127      *   score = | mean(e^{i*ph}) |       (R1, in [0,1])
128      *
129      * For uniformly random event times R1 ≈ 1/sqrt(n) → small.
130      *
131      * Only A events are pushed to the BPH history (one per beat), so a
132      * pure first-harmonic Rayleigh score correctly peaks at the true
133      * beat period.
134      *
135      * Earlier versions of this function returned R1+R2 (sum of first and
136      * second harmonics), motivated by an A+C event stream where two
137      * peaks per period maximized the second harmonic. Once the BPH path
138      * was changed to A-only events, R2 became actively harmful: it
139      * doubles the score for HALF-period candidates because all events
140      * project to phase 0 in the doubled-frequency space. That caused
141      * 18000 BPH watches with intermittent secondary echoes to be
142      * mis-detected as 36000 BPH. Reverted to R1-only in V5.3. */
143     double tg_phase_score(const double *event_times, size_t n, double period) {
144         if (n < 6 || period <= 0.0) return 0.0;
145         double s1 = 0.0, c1 = 0.0;
146         double inv_T = 1.0 / period;
147         for (size_t i = 0; i < n; ++i) {
148             double ph = 2.0 * M_PI * (event_times[i] - floor(event_times[i] * inv_T) * period) * inv_T;
149             c1 += cos(ph);  s1 += sin(ph);
150         }
151         double inv_n = 1.0 / (double)n;
152         return sqrt(c1*c1 + s1*s1) * inv_n;
153     }

155     int tg_bph_pick_by_phase(const double *event_times, size_t n,
156                              const int *list, size_t list_len,
157                              double min_score,
158                              double *out_score,
159                              double *out_period)
160     {
161         if (out_score)  *out_score  = 0.0;
162         if (out_period) *out_period = 0.0;
163         if (n < 6 || !list || list_len == 0) return 0;
164     
165         /* Compute median A-to-A interval. The true beat period is at
166          * least this large (one A per beat means consecutive A's are
167          * separated by ~beat_period; spurious extra events can only
168          * make the median smaller, never larger).
169          *
170          * Test periods less than 0.7 * median are physically implausible
171          * and rejected. This breaks the period/half-period aliasing in
172          * the Rayleigh score (R1 alone can't distinguish T from T/2 for
173          * perfectly periodic events; same for any T/k). */
174         double aa[256];
175         int    n_aa = 0;
176         for (size_t i = 1; i < n && n_aa < 256; ++i) {
177             double d = event_times[i] - event_times[i-1];
178             if (d > 0.0) aa[n_aa++] = d;
179         }
180         /* simple insertion sort for small n_aa */
181         for (int i = 1; i < n_aa; ++i) {
182             double v = aa[i];
183             int j = i - 1;
184             while (j >= 0 && aa[j] > v) { aa[j+1] = aa[j]; j--; }
185             aa[j+1] = v;
186         }
187         double median_aa = (n_aa > 0) ? aa[n_aa / 2] : 0.0;
188         double min_period = 0.7 * median_aa;
189     
190         int best_bph = 0;
191         double best_score = -1.0;
192         double best_period = 0.0;
193         for (size_t i = 0; i < list_len; ++i) {
194             double T = 3600.0 / (double)list[i];
195             if (T < min_period) continue;   /* implausibly short */
196             double s = tg_phase_score(event_times, n, T);
197             if (s > best_score) {
198                 best_score  = s;
199                 best_bph    = list[i];
200                 best_period = T;
201             }
202         }
203         if (out_score)  *out_score  = best_score;
204         if (out_period) *out_period = best_period;
205         if (best_score >= min_score) return best_bph;
206         return 0;
207     }

243     int tg_sync_update(tg_sync_t *s, double event_time) {
244         if (!s->synced) return 0;
245     
246         /* Phase-based matching: each event should fall near one of two phases
247          * within the beat period - phase 0 (reference event) and phase |ac_offset|
248          * (companion event). The A/C direction (which came first) doesn't matter
249          * to the PLL - it just needs the 2-event-per-beat pattern to hold. */
250     
251         /* Compute phase in [0, T). Reference phase is the last matched A's
252          * absolute time mod T (we use next_a_time - beat_period). */
253         double ref = s->next_a_time - s->beat_period;    /* last A time */
254         double phase = event_time - ref;
255         /* Bring into principal range [-T/2, +T/2) for easier comparison. */
256         double T = s->beat_period;
257         while (phase >  0.5 * T) phase -= T;
258         while (phase < -0.5 * T) phase += T;
259     
260         double g = s->ac_offset;   /* magnitude of within-beat gap */
261         /* Expected phases: 0 (next A) or +/- g (C within this or next beat).
262          * ac_offset may be signed; take it as given but accept both signs. */
263         double err_a      = fabs(phase);
264         double err_c_pos  = fabs(phase - g);
265         double err_c_neg  = fabs(phase + g);
266         double err_c      = (err_c_pos < err_c_neg) ? err_c_pos : err_c_neg;
267     
268         if (err_a <= s->tolerance_s) {
269             /* Matched main phase - update reference */
270             s->next_a_time = event_time + s->beat_period;
271             s->beat_period += s->period_gain * phase;
272             s->consecutive_misses = 0;
273             s->last_match_time = event_time;
274             return 1;
275         }
276         if (err_c <= s->tolerance_s) {
277             /* Matched companion phase. Don't advance next_a_time, but refine g. */
278             if (g > 0) {
279                 double nudge = (err_c_pos < err_c_neg) ? (phase - g) : -(phase + g);
280                 s->ac_offset += s->ac_gain * nudge;
281             }
282             s->consecutive_misses = 0;
283             s->last_match_time = event_time;
284             return 1;
285         }
286     
287         /* Event didn't match either expected phase. If we're far past the
288          * predicted next A, advance the window (handles bursts of missed beats). */
289         while (event_time > s->next_a_time + 1.5 * s->beat_period) {
290             s->next_a_time += s->beat_period;
291         }
292     
293         s->consecutive_misses++;
294         if (s->consecutive_misses >= s->max_misses) {
295             s->synced = 0;
296         }
297         return 0;
298     }

300     int tg_sync_check_timeout(tg_sync_t *s, double stream_time) {
301         if (!s->synced) return 0;
302         /* Silence timeout: if the current stream position is more than a few
303          * beat periods past the last matched event, declare sync lost.
304          * Silence is a stronger signal than "events arrive but don't match"
305          * (which gets the full max_misses budget), so we use a tighter
306          * timeout of max(3, max_misses/2) beat periods. */
307         int timeout_beats = s->max_misses / 2;
308         if (timeout_beats < 3) timeout_beats = 3;
309         double timeout = (double)timeout_beats * s->beat_period;
310         if (stream_time - s->last_match_time > timeout) {
311             s->synced = 0;
312             return 1;
313         }
314         return 0;
315     }
