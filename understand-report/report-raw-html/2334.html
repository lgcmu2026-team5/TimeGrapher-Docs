15     /*
16         IMPLEMENTATION OVERVIEW
17         -----------------------
18         The stream maintains an absolute sample counter and a scheduled next event.
19         During watch_synth_stream_fill_f32():
20     
21           1. For each requested output sample, check whether the next Tick/Tock
22              event is due at or before the current absolute sample index.
23           2. If due, create an active packet whose lobes may span many future
24              samples and possibly many future fill calls.
25           3. Mix all active packets at the current sample.
26           4. Apply optional contact/case resonance. Resonator state is stored in the
27              stream object, so ringing continues smoothly across buffer boundaries.
28           5. Add optional mechanical bed noise. The noise filter states are also
29              persistent across buffer boundaries.
30           6. Clip to normalized float PCM range [-1.0, +1.0].
31     
32         The design deliberately keeps event scheduling separate from packet mixing.
33         Event scheduling controls the ground-truth beat timing. Packet mixing only
34         turns each scheduled event into acoustic-looking PCM samples.
35     */

43     /*
44         SplitMix64 pseudo-random generator.
45         It is deterministic, tiny, portable C, and adequate for repeatable synthetic
46         jitter/noise. It is not intended for cryptography.
47     */
48     static uint64_t ws_next_u64(uint64_t *state) {
49         uint64_t z;
50         if (*state == 0) *state = 0x9e3779b97f4a7c15ULL;
51         z = (*state += 0x9e3779b97f4a7c15ULL);
52         z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
53         z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
54         return z ^ (z >> 31);
55     }
56     static double ws_rand01(uint64_t *state) { return (double)(ws_next_u64(state) >> 11) * (1.0 / 9007199254740992.0); }
57     static double ws_rand_signed(uint64_t *state) { return 2.0 * ws_rand01(state) - 1.0; }

171     /*
172         Convert mechanical watch amplitude to synthetic A->C time.
173     
174         The event time is treated as the A-like onset. The strongest C-like lobe is
175         placed a_to_c_time_s later. A detector can therefore compare its measured
176         A and C timestamps against this known ground truth.
177     
178         Formula used when use_watch_amplitude_for_a_to_c is enabled:
179     
180             beat_interval_s = 3600 / BPH
181     
182             A_to_C_seconds =
183                 (2 * beat_interval_s / pi) *
184                 asin(lift_angle_degrees / (2 * watch_amplitude_degrees))
185     
186         Why this is different from the previous version:
187           The previous version used:
188     
189               beat_interval_s * lift_angle_degrees / watch_amplitude_degrees
190     
191           That treats the balance as if it moved linearly through the entire beat
192           interval. That made A->C too large. A balance is better approximated as
193           sinusoidal; it moves fastest around center, so the lift-angle crossing
194           takes a much smaller fraction of the beat.
195     
196         Important behavior:
197           - Higher watch_amplitude_degrees -> shorter A->C time.
198           - Lower watch_amplitude_degrees  -> longer A->C time.
199           - min/max clamps prevent unrealistic or invalid packet geometry.
200     */
201     double watch_synth_stream_compute_a_to_c_time_s(const WatchSynthStreamConfig *cfg) {
202         double t;
203         if (!cfg) return 0.0;
204         if (cfg->use_watch_amplitude_for_a_to_c) {
205             double beat_interval_s = 3600.0 / cfg->bph;
206             double ratio;
207     
208             /*
209                 Sinusoidal balance approximation.
210     
211                 Let:
212                   A = watch_amplitude_degrees
213                   L = lift_angle_degrees
214                   T = beat_interval_s, the time from Tick to Tock
215     
216                 The balance angular position is approximated by:
217                   theta(t) = A * sin(pi * t / T)
218     
219                 Around the zero crossing, the lift zone spans roughly -L/2 to +L/2.
220                 Solving for the time between those two crossings gives:
221     
222                   A_to_C = (2T / pi) * asin(L / (2A))
223     
224                 For small L/(2A), this is approximately:
225                   A_to_C ~= T * L / (pi * A)
226     
227                 This correction is why the spacing is roughly 1/pi of the older
228                 linear formula and is much closer to real timegrapher behavior.
229             */
230             ratio = cfg->lift_angle_degrees / (2.0 * cfg->watch_amplitude_degrees);
231             ratio = ws_clamp(ratio, 0.0, 0.999999);
232             t = (2.0 * beat_interval_s / M_PI) * asin(ratio);
233         } else {
234             t = cfg->manual_a_to_c_time_s;
235         }
236         return ws_clamp(t, cfg->min_a_to_c_time_s, cfg->max_a_to_c_time_s);
237     }

239     /*
240         Validate public configuration before a stream starts.
241     
242         The checks are intentionally conservative. They catch unit mistakes such as:
243           - using 210 instead of 0.210 for beat_error_ms
244           - giving PCM amplitude outside normalized range
245           - asking for a beat error/jitter/wander combination that could produce a
246             non-positive interval at high BPH
247     */
248     bool watch_synth_stream_validate_config(const WatchSynthStreamConfig *cfg, char *err, size_t err_size) {
249         double adjusted, beat_error_offset_s;
250         if (!cfg) { snprintf(err, err_size, "cfg is NULL"); return false; }
251         if (!watch_synth_stream_is_supported_rate(cfg->sample_rate_hz)) { snprintf(err, err_size, "sample_rate_hz must be %u..%u Hz", WS_MIN_SR, WS_MAX_SR); return false; }
252         if (cfg->bph < WS_MIN_BPH || cfg->bph > WS_MAX_BPH) { snprintf(err, err_size, "bph must be %.0f..%.0f", WS_MIN_BPH, WS_MAX_BPH); return false; }
253         if (cfg->pcm_peak_amplitude < 0.0 || cfg->pcm_peak_amplitude > 1.0) { snprintf(err, err_size, "pcm_peak_amplitude must be 0..1 normalized PCM"); return false; }
254         if (cfg->noise_peak_amplitude < 0.0 || cfg->noise_peak_amplitude > 1.0) { snprintf(err, err_size, "noise_peak_amplitude must be 0..1 normalized PCM"); return false; }
255         if (cfg->watch_amplitude_degrees < 90.0 || cfg->watch_amplitude_degrees > 450.0) { snprintf(err, err_size, "watch_amplitude_degrees must be 90..450 degrees"); return false; }
256         if (cfg->lift_angle_degrees <= 0.0 || cfg->lift_angle_degrees > 90.0) { snprintf(err, err_size, "lift_angle_degrees must be >0 and <=90 degrees"); return false; }
257         if (cfg->packet_tail_after_c_s <= 0.0 || cfg->packet_tail_after_c_s > 0.100) { snprintf(err, err_size, "packet_tail_after_c_s must be >0 and <=100 ms"); return false; }
258         if (cfg->post_c_lobe_scale < 0.0 || cfg->post_c_lobe_scale > 2.0) { snprintf(err, err_size, "post_c_lobe_scale must be 0..2"); return false; }
259         if (cfg->c_peak_anchor_gain < 0.0 || cfg->c_peak_anchor_gain > 10.0) { snprintf(err, err_size, "c_peak_anchor_gain must be 0..10"); return false; }
260         if (cfg->c_peak_anchor_width_s <= 0.0 || cfg->c_peak_anchor_width_s > 0.0010) { snprintf(err, err_size, "c_peak_anchor_width_s must be >0 and <=1 ms"); return false; }
261         if (cfg->min_a_to_c_time_s <= 0.0 || cfg->max_a_to_c_time_s <= cfg->min_a_to_c_time_s) { snprintf(err, err_size, "A-to-C clamp range is invalid"); return false; }
262         if (cfg->start_time_s < 0.0) { snprintf(err, err_size, "start_time_s must be >=0"); return false; }
263         adjusted = (3600.0 / cfg->bph) / (1.0 + cfg->rate_error_s_per_day / 86400.0);
264         beat_error_offset_s = fabs(cfg->beat_error_ms) * 1.0e-3;
265         if (adjusted <= beat_error_offset_s + cfg->timing_jitter_us * 1.0e-6 + cfg->bph_wander_depth_us * 1.0e-6) { snprintf(err, err_size, "beat error/jitter/wander too large for BPH"); return false; }
266         return true;
267     }

270         Initialize a caller-owned stream object.
271     
272         No output buffer is retained by the stream. The stream only stores timing,
273         active packet, resonator, noise, and RNG state needed to continue generation
274         on the next fill call.
275     */
276     bool watch_synth_stream_init(WatchSynthStream *s, const WatchSynthStreamConfig *cfg, char *err, size_t err_size) {
277         if (!s) { snprintf(err, err_size, "stream is NULL"); return false; }
278         if (!watch_synth_stream_validate_config(cfg, err, err_size)) return false;
279         memset(s, 0, sizeof(*s));
280         s->cfg = *cfg;
281         s->rng_state = cfg->seed ? cfg->seed : 0x123456789abcdefULL;
282         s->adjusted_interval_s = (3600.0 / cfg->bph) / (1.0 + cfg->rate_error_s_per_day / 86400.0);
283         watch_synth_stream_reset(s);
284         return true;
285     }

341     /*
342         Add one damped sinusoidal lobe to a packet, with optional variation.
343     
344         A lobe is the building block for the synthetic acoustic packet:
345           delay_s : time after A/onset when the lobe starts
346           amp     : relative contribution before packet_gain scaling
347           freq    : ringing frequency in Hz
348           tau     : exponential decay time constant in seconds
349     
350         Realistic mode perturbs delay/frequency/decay/level per packet so every tick
351         is not an exact copy of the previous one.
352     */
353     static void ws_add_varied_lobe(WatchSynthStream *s, WatchSynthActivePacket *p, double packet_duration_s, double delay_s, double amp, double freq, double tau, double freq_scale) {
354         const WatchSynthStreamConfig *cfg = &s->cfg;
355         freq *= freq_scale;
356         if (cfg->enable_packet_shape_variation) {
357             delay_s += cfg->shape_delay_jitter_us * 1.0e-6 * ws_rand_signed(&s->rng_state);
358             amp *= 1.0 + 0.16 * ws_rand_signed(&s->rng_state);
359             freq *= 1.0 + cfg->shape_frequency_jitter * ws_rand_signed(&s->rng_state);
360             tau *= 1.0 + cfg->shape_decay_jitter * ws_rand_signed(&s->rng_state);
361         }
362         delay_s = ws_clamp(delay_s, 0.0, packet_duration_s - 0.0001);
363         amp = ws_clamp(amp, 0.0, 2.0);
364         freq = ws_clamp(freq, 500.0, 0.45 * (double)cfg->sample_rate_hz);
365         tau = ws_clamp(tau, 0.00015, 0.020);
366         ws_add_lobe(p, delay_s, amp, freq, tau);
367     }

369     /*
370         Schedule the next Tick/Tock event.
371     
372         The just-emitted event's kind determines which half interval follows:
373           Tick -> Tock uses +beat_error_ms
374           Tock -> Tick uses -beat_error_ms
375     
376         beat_error_ms is converted to seconds here. The code also adds optional
377         independent timing jitter and optional slow BPH wander.
378     */
379     static void ws_schedule_next_event(WatchSynthStream *s, WatchSynthEventKind kind) {
380         const WatchSynthStreamConfig *cfg = &s->cfg;
381         double beat_error_offset_s = cfg->beat_error_ms * 1.0e-3;
382         double offset_s = (kind == WATCH_SYNTH_EVENT_TICK) ? +beat_error_offset_s : -beat_error_offset_s;
383         double jitter_s = cfg->timing_jitter_us * 1.0e-6 * ws_rand_signed(&s->rng_state);
384         if (cfg->enable_bph_wander) {
385             s->current_bph_wander_us += cfg->bph_wander_step_us * ws_rand_signed(&s->rng_state);
386             s->current_bph_wander_us = ws_clamp(s->current_bph_wander_us, -cfg->bph_wander_depth_us, cfg->bph_wander_depth_us);
387         } else s->current_bph_wander_us = 0.0;
388         s->next_interval_offset_us = offset_s * 1.0e6;
389         s->next_interval_jitter_us = jitter_s * 1.0e6;
390         s->next_event_time_s += s->adjusted_interval_s + offset_s + jitter_s + s->current_bph_wander_us * 1.0e-6;
391         s->next_event_sample_index = (uint64_t)llround(s->next_event_time_s * (double)cfg->sample_rate_hz);
392     }

394     /*
395         Start a new acoustic packet and return its ground-truth event record.
396     
397         The generated event time is the A/onset time. The packet includes early
398         A-like lobes, optional middle impacts, and one or more C-like lobes near the
399         computed A->C time. In realistic mode, the C zone is a small cluster rather
400         than a single perfect sinusoid.
401     */
402     static WatchSynthStreamEvent ws_start_packet(WatchSynthStream *s) {
403         const WatchSynthStreamConfig *cfg = &s->cfg;
404         WatchSynthActivePacket *p = ws_alloc_packet(s);
405         WatchSynthStreamEvent e;
406         WatchSynthEventKind kind = (s->beat_index & 1u) ? WATCH_SYNTH_EVENT_TOCK : WATCH_SYNTH_EVENT_TICK;
407         double a_to_c_s = watch_synth_stream_compute_a_to_c_time_s(cfg);
408         double packet_duration_s = a_to_c_s + cfg->packet_tail_after_c_s;
409         uint64_t packet_samples = (uint64_t)ceil(packet_duration_s * (double)cfg->sample_rate_hz) + 2u;
410         double gain = cfg->pcm_peak_amplitude;
411         double freq_scale = 1.0, c_gain = 1.0;
412     
413         if (cfg->packet_gain_variation > 0.0) gain *= 1.0 + cfg->packet_gain_variation * ws_rand_signed(&s->rng_state);
414         if (cfg->enable_amplitude_drift) {
415             double phase = 2.0 * M_PI * s->next_event_time_s / cfg->amplitude_drift_period_s;
416             gain *= 1.0 + cfg->amplitude_drift_depth * sin(phase);
417         }
418         gain = ws_clamp(gain, 0.0, 1.5);
419     
420         memset(p, 0, sizeof(*p));
421         p->active = 1;
422         p->start_sample_index = s->next_event_sample_index;
423         p->end_sample_index = p->start_sample_index + packet_samples;
424         p->kind = kind;
425         p->polarity = kind == WATCH_SYNTH_EVENT_TICK ? 1.0 : -0.94;
426         p->packet_gain = gain;
427     
428         if (cfg->enable_tick_tock_spectral_diff) {
429             if (kind == WATCH_SYNTH_EVENT_TICK) { freq_scale = 1.04; c_gain = 1.00; }
430             else { freq_scale = 0.93; c_gain = 0.91; }
431         }
432         if (cfg->enable_realistic_packet) {
433             /* A-like onset cluster: early impulse/fork activity near event time. */
434             ws_add_varied_lobe(s,p,packet_duration_s,0.00000,0.22,2300.0,0.00085,freq_scale);
435             ws_add_varied_lobe(s,p,packet_duration_s,0.00028,0.18,4100.0,0.00070,freq_scale);
436             ws_add_varied_lobe(s,p,packet_duration_s,0.00072,0.13,6800.0,0.00055,freq_scale);
437     
438             /* Middle low-energy impacts/rattle before the main C zone. */
439             ws_add_varied_lobe(s,p,packet_duration_s,0.00215,0.16,5200.0,0.00085,freq_scale);
440             ws_add_varied_lobe(s,p,packet_duration_s,0.00305,0.10,8800.0,0.00060,freq_scale);
441     
442             /*
443                 C-like locking/banking cluster.
444     
445                 With C-peak lock enabled, a narrow Gaussian C anchor is placed
446                 exactly at the computed A->C time. The surrounding ringing lobes
447                 remain present but are deliberately lower, so a later C-ring peak
448                 should not steal the amplitude measurement at 192 kHz or higher.
449             */
450             if (cfg->enable_c_peak_lock) {
451                 double post = cfg->post_c_lobe_scale;
452                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s-0.00105,0.12*c_gain,4700.0,0.00065,freq_scale);
453                 ws_set_c_anchor(p, a_to_c_s, cfg->c_peak_anchor_gain * c_gain, cfg->c_peak_anchor_width_s);
454                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00072,0.18*post*c_gain,10300.0,0.00065,freq_scale);
455                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00175,0.10*post*c_gain,6100.0,0.00090,freq_scale);
456                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00320,0.05*post*c_gain,2900.0,0.00150,freq_scale);
457             } else {
458                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s-0.00105,0.34*c_gain,4700.0,0.00110,freq_scale);
459                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s,1.00*c_gain,7600.0,0.00185,freq_scale);
460                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00062,0.53*c_gain,10300.0,0.00120,freq_scale);
461                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00170,0.24*c_gain,6100.0,0.00180,freq_scale);
462                 ws_add_varied_lobe(s,p,packet_duration_s,a_to_c_s+0.00320,0.12*c_gain,2900.0,0.00300,freq_scale);
463             }
464         } else {
465             ws_add_lobe(p, 0.0000, 0.38, 2800.0, 0.0013);
466             if (cfg->enable_c_peak_lock) {
467                 ws_set_c_anchor(p, a_to_c_s, cfg->c_peak_anchor_gain, cfg->c_peak_anchor_width_s);
468                 ws_add_lobe(p, a_to_c_s + 0.0008, 0.10 * cfg->post_c_lobe_scale, 9800.0, 0.0008);
469             } else {
470                 ws_add_lobe(p, a_to_c_s, 1.00, 7600.0, 0.0018);
471                 ws_add_lobe(p, a_to_c_s + 0.0007, 0.36, 9800.0, 0.0010);
472             }
473         }
474     
475         memset(&e, 0, sizeof(e));
476         e.beat_index = s->beat_index;
477         e.kind = kind;
478         e.time_s = s->next_event_time_s;
479         e.sample_index = s->next_event_sample_index;
480         e.packet_gain = gain;
481         e.a_to_c_time_s = a_to_c_s;
482         e.watch_amplitude_degrees = cfg->watch_amplitude_degrees;
483         e.lift_angle_degrees = cfg->lift_angle_degrees;
484         e.bph_wander_us = s->current_bph_wander_us;
485         if (s->beat_index > 0) {
486             e.interval_from_previous_us = (s->next_event_time_s - s->last_event_time_s) * 1.0e6;
487             e.applied_interval_offset_us = s->next_interval_offset_us;
488             e.timing_jitter_us = s->next_interval_jitter_us;
489         }
490         s->last_event_time_s = s->next_event_time_s;
491         ++s->beat_index;
492         ws_schedule_next_event(s, kind);
493         return e;
494     }

496     static double ws_packet_sample(WatchSynthActivePacket *p, uint64_t abs_sample, double sr) {
497         size_t i; double y = 0.0;
498         if (!p->active) return 0.0;
499         if (abs_sample < p->start_sample_index) return 0.0;
500         if (abs_sample >= p->end_sample_index) { p->active = 0; return 0.0; }
501         for (i = 0; i < p->lobe_count; ++i) {
502             const WatchSynthLobe *l = &p->lobes[i];
503             double rel_s = (double)(abs_sample - p->start_sample_index) / sr - l->delay_s;
504             if (rel_s >= 0.0) {
505                 double attack = rel_s < 0.00012 ? rel_s / 0.00012 : 1.0;
506                 double env = exp(-rel_s / l->tau_s);
507                 y += p->polarity * p->packet_gain * l->rel_amp * attack * env * sin(2.0 * M_PI * l->freq_hz * rel_s);
508             }
509         }
510         if (p->c_anchor_enabled) {
511             double rel_s = (double)(abs_sample - p->start_sample_index) / sr - p->c_anchor_delay_s;
512             double w = p->c_anchor_width_s > 0.0 ? p->c_anchor_width_s : 0.000020;
513             double g = exp(-0.5 * (rel_s / w) * (rel_s / w));
514             y += p->polarity * p->packet_gain * p->c_anchor_gain * g;
515         }
516         return y;
517     }

519     static double ws_resonator(double x, double *s1, double *s2, double sr, double f, double q, double gain) {
520         double bw, r, w, a1, a2, st;
521         if (gain == 0.0 || f <= 0.0 || q <= 0.0) return x;
522         f = ws_clamp(f, 20.0, 0.45 * sr); bw = f / q; r = exp(-M_PI * bw / sr); w = 2.0 * M_PI * f / sr;
523         a1 = 2.0 * r * cos(w); a2 = -(r * r); st = x + a1 * (*s1) + a2 * (*s2); *s2 = *s1; *s1 = st;
524     
525         /*
526             Normalize resonator contribution by (1-r). Without this, high sample
527             rates make r very close to 1 and the recursive state can grow enough to
528             clip, creating artificial late peaks.
529         */
530         return x + gain * (1.0 - r) * st;
531     }
532     static double ws_lp(double x, double *state, double cutoff, double sr) { double a = exp(-2.0 * M_PI * ws_clamp(cutoff, 1.0, 0.45*sr) / sr); *state = (1.0-a)*x + a*(*state); return *state; }
533     static double ws_hp(double x, double *lp_state, double cutoff, double sr) { return x - ws_lp(x, lp_state, cutoff, sr); }
534     static double ws_noise(WatchSynthStream *s) {
535         const WatchSynthStreamConfig *cfg = &s->cfg; double n;
536         if (cfg->noise_peak_amplitude <= 0.0) return 0.0;
537         n = cfg->noise_peak_amplitude * ws_rand_signed(&s->rng_state);
538         if (cfg->enable_bandlimited_noise) { n = ws_hp(n, &s->noise_hp_low_state, cfg->noise_low_hz, (double)cfg->sample_rate_hz); n = ws_lp(n, &s->noise_lp_state, cfg->noise_high_hz, (double)cfg->sample_rate_hz); }
539         return n;
540     }

542     WatchSynthStreamFillResult watch_synth_stream_fill_f32(WatchSynthStream *s, float *out_pcm, size_t out_count, WatchSynthStreamEvent *events, size_t event_capacity) {
543         WatchSynthStreamFillResult r; size_t i; double sr;
544         memset(&r, 0, sizeof(r));
545         if (!s || !out_pcm || out_count == 0) return r;
546         r.first_sample_index = s->absolute_sample_index; sr = (double)s->cfg.sample_rate_hz;
547         for (i = 0; i < out_count; ++i) {
548             uint64_t abs_sample = s->absolute_sample_index; double y = 0.0; size_t pidx;
549             while (s->next_event_sample_index <= abs_sample) {
550                 WatchSynthStreamEvent e = ws_start_packet(s);
551                 if (events) {
552                     if (r.events_written < event_capacity) events[r.events_written++] = e;
553                     else r.events_dropped++;
554                 }
555             }
556             for (pidx = 0; pidx < WATCH_SYNTH_MAX_ACTIVE_PACKETS; ++pidx) y += ws_packet_sample(&s->active_packets[pidx], abs_sample, sr);
557             if (s->cfg.enable_sensor_resonance) {
558                 y = ws_resonator(y, &s->resonator1_s1, &s->resonator1_s2, sr, s->cfg.sensor_resonance1_hz, s->cfg.sensor_resonance1_q, s->cfg.sensor_resonance1_gain);
559                 y = ws_resonator(y, &s->resonator2_s1, &s->resonator2_s2, sr, s->cfg.sensor_resonance2_hz, s->cfg.sensor_resonance2_q, s->cfg.sensor_resonance2_gain);
560             }
561             y += ws_noise(s);
562             out_pcm[i] = (float)ws_clamp(y, -1.0, 1.0);
563             ++s->absolute_sample_index;
564         }
565         r.samples_written = out_count; r.next_sample_index = s->absolute_sample_index;
566         return r;
567     }
