33     void TAudioWorker::StartAudioRecording(QAudioDevice InputDevice,int SampleRate,float Volume)
34     {
35         QAudioFormat InputFormat;
36         InputFormat.setSampleRate(SampleRate);
37         InputFormat.setChannelCount(CHANNELS);
38         InputFormat.setSampleFormat(SAMPLE_FORMAT);
39     
40         if (mAudioInput) delete mAudioInput;
41         mAudioInput = nullptr;
42     
43         if (InputDevice.isNull()) {
44             qWarning() << "No default audio input device found.";
45             emit finished();
46             return;
47         }
48     
49         mAudioInput = new QAudioSource(InputDevice,InputFormat,this);
50         mAudioInput->setVolume(Volume);
51         connect(mAudioInput, &QAudioSource::stateChanged,this, &TAudioWorker::stateChangeAudioInput);
52         mAudioInputDevice = mAudioInput->start(); // Start recording
53         connect( mAudioInputDevice, &QIODevice::readyRead, this, &TAudioWorker::ProcessAudioInput);
54         qDebug() << "Audio recording started in worker thread.";
55     }

62     void TAudioWorker::ProcessAudioInput()
63     {
64         if (!TimerStarted)
65         {
66             TimerStarted=true;
67             Timer.start();
68         }
69         double CurrentTime;
70     
71         QByteArray ba =  mAudioInputDevice->readAll();
72     
73         unsigned int NumberOfSamples = ba.length() / SAMPLE_SIZE;
74         float *AudioSamples=(float *)ba.constData();
75         mRawAudio->Mutex.lock();
76         unsigned int TempWriteIndex = mRawAudio->WriteIndex;
77         mRawAudio->Mutex.unlock();
78         int SamplesLeft=std::min(NumberOfSamples,mRawAudio->NumberOfAudioSamples-TempWriteIndex);
79         memcpy(&mRawAudio->Samples[TempWriteIndex], AudioSamples, SamplesLeft * SAMPLE_SIZE);
80         //qInfo() << "Bytes in "<< ba.length();
81         if(SamplesLeft < NumberOfSamples)
82         {
83             memcpy(mRawAudio->Samples, &AudioSamples[SamplesLeft], (NumberOfSamples - SamplesLeft) * SAMPLE_SIZE);
84             qInfo() << "MasterAudioData Samples Rollover";
85         }
86         mRawAudio->Mutex.lock();
87         mRawAudio->WriteIndex = (TempWriteIndex+ NumberOfSamples) % mRawAudio->NumberOfAudioSamples;
88         mRawAudio->TotalSamplesWritten+=NumberOfSamples;
89         mRawAudio->Mutex.unlock();
90         ++FrameCount;
91         SampleCount+=NumberOfSamples;
92         CurrentTime = Timer.elapsed()/1000.0;
93         if (CurrentTime-LastTime > 2) // average fps over 2 seconds
94         {
95             double fdelta;
96             fdelta=CurrentTime-LastTime;
97             mRawAudio->FPS=FrameCount/fdelta;
98             mRawAudio->SPS=SampleCount/fdelta;
99             mRawAudio->SPF=SampleCount/FrameCount;
100             LastTime=CurrentTime;
101             FrameCount=0;
102             SampleCount=0;
103         }
104         //qDebug() << "worker thread: handleResults slot is running in thread" << QThread::currentThreadId()<<" "<<count;
105         emit AudioDataReady(); // Emit data to the main thread
106     
107     }
