1     /* Bph.h - BPH lists, phase-score auto-detection, and sync state.
2      *
3      * Three concerns covered by this header:
4      *
5      *   1. Candidate BPH lists. TG_AUTO_BPH_LIST is the small set used by
6      *      auto-detection (the BPHs we'll actually try to lock onto);
7      *      TG_MANUAL_BPH_LIST is the wider set the host UI shows for
8      *      user-driven manual selection (covers many antique rates).
9      *
10      *   2. Phase-score period detection. tg_phase_score computes the
11      *      magnitude of the Rayleigh circular mean of (event_time mod T)
12      *      for a candidate period T, returning a 0..1 measure of phase
13      *      alignment. tg_bph_pick_by_phase sweeps the candidate list and
14      *      picks the best-scoring period, returning 0 if no candidate
15      *      crosses the lock threshold.
16      *
17      *      V5.3 fix: tg_phase_score returns R1 (first harmonic) only.
18      *      Earlier versions returned R1+R2 which caused half-period
19      *      mis-detection on signals where pre-sync events came in pairs
20      *      (the second harmonic doubled the score for half-period
21      *      candidates). See Bph.cpp and docs/design.md sec 4 for the
22      *      full discussion.
23      *
24      *   3. tg_sync_t -- a small PLL-ish tracker that, once a BPH is
25      *      picked, predicts when each next A and C should arrive based
26      *      on the locked beat period and within-beat A->C offset. Each
27      *      incoming event is matched against the prediction window;
28      *      misses are counted, and after max_misses consecutive misses
29      *      sync is declared lost. Both the period and the A->C offset
30      *      adapt slowly (PLL-style) to track real drift over the life
31      *      of the recording.
32      *
33      * V5.5: file renamed from bph.h to Bph.h and used by Bph.cpp;
34      * compiles under C++17. extern "C" in Timegrapher.h keeps the public
35      * symbols ABI-stable for non-C++ callers. */

61     /* Estimate beat period (seconds) from a sequence of event times.
62      * Uses median of t[i+2] - t[i] which equals one full beat period when
63      * events alternate A,C,A,C,...
64      * Returns 0.0 if not enough events. */
65     double tg_estimate_beat_period(const double *event_times, size_t n);
66     
67     /* Score how well event times fit a candidate beat period using the 1st and
68      * 2nd-harmonic Rayleigh statistics on the phase histogram (events folded
69      * mod T). Returns 0..2; > ~0.7 indicates a real fit. Robust to missed
70      * events because each event contributes independently. */
71     double tg_phase_score(const double *event_times, size_t n, double period);
72     
73     /* From a list of candidate BPH values, pick the one whose period maximizes
74      * the phase score on the events. Returns 0 if best score below threshold or
75      * not enough events. min_score recommended ~0.7. */
76     int tg_bph_pick_by_phase(const double *event_times, size_t n,
77                              const int *list, size_t list_len,
78                              double min_score,
79                              double *out_score,
80                              double *out_period);

82     /* Sync state tracker. Tracks expected next event time using the current beat
83      * period and within-beat A-C offset. */
84     typedef struct {
85         int      bph;             /* current locked BPH                         */
86         double   beat_period;     /* seconds                                    */
87         double   ac_offset;       /* mean A->C delta (seconds)                  */
88     
89         /* PLL-ish tracking */
90         double   next_a_time;     /* predicted absolute time of next A          */
91         int      synced;
92         int      consecutive_misses;
93     
94         /* tolerance */
95         double   tolerance_s;     /* +/- around predicted event                 */
96         int      max_misses;
97     
98         /* PLL gains */
99         double   period_gain;     /* default 0.01                               */
100         double   ac_gain;         /* default 0.05                               */
101     
102         /* timestamp of the most recent matched event - used for time-based
103          * sync loss (if current stream time advances far beyond this + miss
104          * budget * beat_period, declare loss). */
105         double   last_match_time;
106     } tg_sync_t;

122     /* Feed a single event. Updates internal model.
123      * Returns:
124      *   1 if it matches the predicted A or C window (sync maintained)
125      *   0 if it doesn't match (a miss)
126      * If consecutive_misses >= max_misses, sets synced=0; caller checks. */
127     int tg_sync_update(tg_sync_t *s, double event_time);

131      * tg_process() iteration after feeding all batch events. Returns 1 if it
132      * cleared synced to 0, else 0. */
133     int tg_sync_check_timeout(tg_sync_t *s, double stream_time);
