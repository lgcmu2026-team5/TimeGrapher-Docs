1     /* Timegrapher.h - V5.0 detection layer.
2      *
3      * libtimegrapher provides ONLY signal detection and BPH/sync tracking
4      * for mechanical-watch audio. Per-event amplitude, predicted-time gate,
5      * autocorrelation, and other analytics live in companion libraries
6      * (libtgan, libtgacf) that consume the events emitted here.
7      *
8      * Streaming pipeline:
9      *
10      *     PCM (float32 mono)
11      *       -> DC blocker (200 Hz HPF)
12      *       -> envelope (full-wave rectify + 0.15 ms LPF)
13      *       -> silence-based onset detector
14      *       -> A and C event timestamps with sub-sample accuracy
15      *       -> Rayleigh phase-score BPH detection + PLL sync tracker
16      *
17      * Output: tg_event_t carrying type, time, sample index, peak value,
18      *         and is_pre_sync flag. NO amplitude, NO predicted time,
19      *         NO parity, NO gate decision -- those are analytics.
20      */

22     #define TIMEGRAPHER_H

50     /* V5.4: C-event timing placement.
51      *
52      * The detector finds a C event in two stages: the C cluster's peak
53      * (envelope max past the C-search-skip window) and, optionally, the
54      * C cluster's onset (rising-edge crossing of half the peak's height,
55      * found by walking back from the peak).
56      *
57      * c_placement controls which timing fills the primary fields of
58      * tg_event_t (sample_index, sub_sample_offset, time_seconds). The
59      * other timing is always available as metadata via the onset_*
60      * fields when found.
61      *
62      * PEAK (default, V5.0-V5.3 behavior): primary timing is the C peak.
63      * ONSET: primary timing is the C onset; falls back to peak if onset
64      *        detection failed for that beat.
65      */
66     typedef enum {
67         TG_C_PLACEMENT_PEAK  = 0,
68         TG_C_PLACEMENT_ONSET = 1
69     } tg_c_placement_t;

105     /* ============================ events ============================== */
106     
107     typedef struct {
108         /* Primary timing (chosen per c_placement for C events; for A
109          * events this is always the onset). */
110         double           time_seconds;     /* sub-sample accurate            */
111         uint64_t         sample_index;     /* integer absolute sample index  */
112         double           sub_sample_offset;/* in [-0.5, +0.5]                */
113         float            peak_value;       /* envelope value at peak         */
114         tg_event_type_t  type;
115         int              is_pre_sync;      /* 1 if before BPH lock           */
116     
117         /* V5.4: C-event onset metadata. Populated for C events when the
118          * backward-walk algorithm successfully locates the C cluster's
119          * rising edge. Zero / 0 for A events and for C events where
120          * onset detection failed (e.g. very noisy signals).
121          *
122          * onset_time_seconds == (onset_sample_index + onset_sub_sample_offset) / sample_rate.
123          *
124          * In TG_C_PLACEMENT_ONSET mode, the primary timing fields above
125          * carry the same value as the onset_* fields here (i.e. onset is
126          * promoted to primary; the original peak is no longer reported).
127          * In TG_C_PLACEMENT_PEAK mode, the primary fields carry the peak
128          * and these onset_* fields are populated as metadata. */
129         uint64_t         onset_sample_index;
130         double           onset_sub_sample_offset;
131         double           onset_time_seconds;
132         int              onset_valid;      /* 1 if onset fields are populated */
133     } tg_event_t;

137     typedef struct {

142     
143         /* Events emitted in THIS call. Pointer is owned by the context;
144          * remains valid until the next tg_process / tg_flush / tg_destroy. */

147     
148         /* Envelope, delayed for alignment with events. processed_pcm[0]
149          * corresponds to absolute input sample processed_pcm_start_sample. */
150         float            *processed_pcm;

152         uint64_t          processed_pcm_start_sample;
153     
154         /* One-shot edge flags */
155         int               sync_lost_event;
156         int               sync_acquired_event;
157         /* V5.6: set for one tg_process call when the detector observed
158          * a large amplitude jump (>= 10x baseline) and the library
159          * flushed adaptive state to recover from a polluted prelude
160          * (e.g. ambient noise before the watch arrives on the mic). */
161         int               detector_reset_event;
162     
163         /* Detector state (instantaneous) for diagnostics / UI */
164         float             onset_threshold;
165         float             min_peak_threshold;
166         float             noise_floor;
167         float             reference_peak;
168     } tg_result_t;
