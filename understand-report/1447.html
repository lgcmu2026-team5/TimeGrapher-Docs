1     /* Detector.cpp -- silence-based burst detector.
2      *
3      * See Detector.h for the full algorithm description and field-by-field
4      * struct documentation. In brief:
5      *
6      *   1. An envelope sample in silence goes into a 256-slot ring buffer
7      *      (one sample per ms); the 75th percentile of that buffer is the
8      *      effective noise floor. This is robust to occasional spikes from
9      *      microphone bumps or background noise.
10      *
11      *   2. The reference peak level is the median of the last 16 accepted
12      *      burst peaks. Median (not mean) so a single anomalously loud or
13      *      quiet beat doesn't move the working threshold.
14      *
15      *   3. Threshold = noise_floor + fraction * (ref_peak - noise_floor)
16      *      with fractions 3% (onset), 2% (release), 20% (min-peak). The
17      *      onset fraction is intentionally low because watch impulses
18      *      have sharp leading edges -- we want to catch them as early as
19      *      possible, then the higher min-peak fraction screens out
20      *      anything that didn't actually rise into a real impulse.
21      *
22      *   4. State machine: SILENCE <-> BURST. Transitions gated by
23      *      wall-clock time since previous burst ended (silence gate) and
24      *      since current burst peak was seen (burst-end gate). This makes
25      *      the detector robust to noisy envelopes that wobble across a
26      *      threshold many times -- only sustained crossings count.
27      *
28      *   5. On accepted burst, emits A (onset-crossing time, linear
29      *      interpolation between the sample below threshold and the first
30      *      above) and C (peak time, parabolic interpolation across the
31      *      maximum and its two neighbors). A is deferred until burst end
32      *      because we need to know the burst's max amplitude before we
33      *      can trust it to be a real beat (rejecting spurious crossings
34      *      that never grew into a full impulse).
35      *
36      *   6. V5.2 C-search-skip: within an accepted burst, the C peak is
37      *      sought only past a small skip window from burst start, so we
38      *      don't mis-place C onto the A's own decay tail or onto an
39      *      intermediate sub-impulse. The skip is 3% of the beat period
40      *      once BPH is known, falling back to 3 ms pre-sync.
41      *
42      *   7. V5.4 C-onset detection: at burst-emission time, walk backward
43      *      from the C peak in a small envelope ring buffer to find the
44      *      half-height crossing on the C cluster's rising edge. Two
45      *      safeguards keep the walk reliable: a minimum below-threshold
46      *      dwell (so within-cluster ringing notches don't register as
47      *      onsets) and a search bound (so the walk can't cross out of
48      *      the C cluster into A territory). The result is attached to
49      *      the emitted event in onset_* fields. Whether onset or peak is
50      *      reported as the primary timing depends on c_placement mode
51      *      handled in Timegrapher.cpp.
52      *
53      * V5.5: file renamed from detector.c to Detector.cpp; compiles under
54      * C++17. No algorithmic change. The detector is plain procedural code
55      * (no constructors, no exceptions, no STL) so the C->C++ rename is
56      * essentially cosmetic at the source level.
57      *
58      * V5.6: regime-change detector. A new short ring (regime_peak_ring,
59      * 8 entries) tracks recent burst peaks. When a new peak is >= 10x
60      * the rolling minimum AND both are above 0.001, the detector sets
61      * regime_reset_pending. The library reads-and-clears this flag via
62      * tg_detector_consume_regime_reset and flushes adaptive state to
63      * recover from polluted preludes (e.g. ambient noise before the
64      * watch arrives on the microphone). One reset per second max via
65      * an internal cooldown using total_samples + regime_last_reset_idx.
66      */

134     void tg_detector_init(tg_detector_t *d, double fs) {
135         d->fs = fs;
136     
137         /* EMA time constants used only for bootstrap fallbacks (until the
138          * percentile and median buffers have enough samples). */
139         d->noise_alpha = 1.0 - exp(-1.0 / (0.3 * fs));  /* tau 0.3 s */
140         d->ceil_alpha  = 1.0 - exp(-1.0 / (1.5 * fs));  /* tau 1.5 s */
141     
142         d->warmup_samples = (uint64_t)(0.2 * fs);  /* skip first 200 ms */
143     
144         /* Pre-sync defaults:
145          *   min_silence = 20 ms (minimum time between beats before lock)
146          *   burst_end   = 10 ms (minimum time past peak before emitting C)
147          * These get tightened once BPH is locked; see
148          * tg_detector_set_min_silence(). */
149         d->min_silence_samples = (uint64_t)(0.020 * fs);
150         d->burst_end_samples   = (uint64_t)(0.010 * fs);
151     
152         /* Sample the envelope into the noise-percentile buffer once per ~1 ms
153          * during silence. 256 slots -> ~250 ms of recent silence history. */
154         d->noise_sample_interval = (uint64_t)(0.001 * fs);
155         if (d->noise_sample_interval < 1) d->noise_sample_interval = 1;
156     
157         /* Default gate fractions. These are the scale-invariant knobs that
158          * determine detection sensitivity. */
159         d->onset_fraction    = 0.03;
160         d->min_peak_fraction = 0.20;
161     
162         /* V4.5: A-to-A interval gate disabled at init / pre-sync. Library
163          * sets it once BPH is known via tg_detector_set_min_a_interval. */
164         d->min_a_interval_samples = 0;
165     
166         /* V5.2: C-search skip default. 3 ms covers the common multi-impulse
167          * "first sub-impulse spike" cases while staying below physical
168          * t_AC_min for any reasonable amplitude. Library may tighten this
169          * with tg_detector_set_c_search_skip() once BPH is known. */
170         d->c_search_skip_samples = (uint64_t)(0.003 * fs);
171     
172         /* V5.4: C-onset detection defaults.
173          * Search bound 5 ms pre-sync (library tightens to t_AC_min/2 once
174          * BPH is known). Dwell 0.3 ms below half-height to skip
175          * within-cluster ringing notches. */
176         d->c_onset_search_max_samples = (uint64_t)(0.005 * fs);
177         d->c_onset_dwell_samples      = (uint64_t)(0.0003 * fs);
178         if (d->c_onset_dwell_samples < 2) d->c_onset_dwell_samples = 2;
179     
180         /* Allocate the recent-envelope ring buffer. Size to cover:
181          *   - burst_end_samples (~25 ms post-peak before emit) — this
182          *     is when find_c_onset runs, so peak is already that old
183          *   - the backward search window itself (up to 5 ms)
184          *   - margin for the burst's pre-peak portion
185          *
186          * 50 ms covers all known watch BPHs with comfortable headroom.
187          * Cost: ~2.4 KB at 48 kHz, ~19 KB at 384 kHz. */
188         if (d->env_ring) {
189             free(d->env_ring);
190             d->env_ring = NULL;
191         }
192         d->env_ring_capacity = (size_t)(0.050 * fs);
193         if (d->env_ring_capacity < 256) d->env_ring_capacity = 256;
194         d->env_ring          = (float*)calloc(d->env_ring_capacity, sizeof(float));
195         d->env_ring_head     = 0;
196         d->env_ring_newest_abs = 0;
197         d->env_ring_has_data   = 0;
198     
199         /* V5.6: regime-change detector. last_reset_idx is set once at init
200          * to 0 (meaning "no reset has happened yet"). tg_detector_reset
201          * does not clear this -- it persists across resets to enforce
202          * the cooldown gate, so a reset can't be triggered again
203          * immediately. */
204         d->regime_last_reset_idx = 0;
205     
206         tg_detector_reset(d);
207     }

226     void tg_detector_set_c_onset_search_max(tg_detector_t *d, double max_s) {
227         if (!d) return;
228         if (max_s < 0.0) max_s = 0.0;
229         uint64_t s = (uint64_t)(max_s * d->fs);
230         /* Cap to ring capacity - dwell margin so the walk can't run off
231          * the buffer's tail. */
232         if (s + d->c_onset_dwell_samples >= d->env_ring_capacity) {
233             s = (d->env_ring_capacity > d->c_onset_dwell_samples + 1)
234                     ? (d->env_ring_capacity - d->c_onset_dwell_samples - 1)
235                     : 0;
236         }
237         d->c_onset_search_max_samples = s;
238     }

267     static inline double parabolic_offset(double y_m1, double y_0, double y_p1) {
268         double denom = (y_m1 - 2.0 * y_0 + y_p1);
269         if (fabs(denom) < 1e-20) return 0.0;
270         double off = 0.5 * (y_m1 - y_p1) / denom;
271         if (off < -0.5) off = -0.5;
272         if (off >  0.5) off =  0.5;
273         return off;
274     }

276     /* V5.4: read the envelope sample at absolute index `abs` from the
277      * ring buffer. The ring stores the most recent `env_ring_capacity`
278      * samples ending at `env_ring_newest_abs` (the abs index of the last
279      * sample written). Returns 0 if `abs` is out of range. */
280     static inline double env_ring_at(const tg_detector_t *d, uint64_t abs) {
281         if (!d->env_ring || d->env_ring_capacity == 0) return 0.0;
282         if (!d->env_ring_has_data)                     return 0.0;
283         uint64_t newest_abs = d->env_ring_newest_abs;
284         if (abs > newest_abs) return 0.0;
285         uint64_t age = newest_abs - abs;
286         if (age >= d->env_ring_capacity) return 0.0;
287         /* The most-recent slot is at (head - 1) mod cap. */
288         size_t newest_slot = (d->env_ring_head + d->env_ring_capacity - 1)
289                                 % d->env_ring_capacity;
290         size_t idx = (newest_slot + d->env_ring_capacity - age)
291                                 % d->env_ring_capacity;
292         return (double)d->env_ring[idx];
293     }

295     /* V5.4: find the C-cluster onset by walking backward from the C peak.
296      *
297      * Algorithm:
298      *   threshold = 0.5 * c_peak_value
299      *   walk backward up to search_max samples
300      *     count consecutive samples with envelope < threshold
301      *     when count >= dwell_samples, the onset is the last sample
302      *       (in time) where envelope was >= threshold
303      *
304      * Returns 1 with onset_idx_out / onset_sub_off_out filled in on
305      * success, 0 if onset wasn't found (envelope never dipped sufficiently
306      * before peak, or search ran past the bound).
307      *
308      * onset_sub_off_out is a linear-interpolation refinement on the
309      * threshold crossing, in [-0.5, +0.5].
310      */
311     static int find_c_onset(const tg_detector_t *d,
312                             uint64_t  c_peak_idx,
313                             double    c_peak_value,
314                             uint64_t *onset_idx_out,
315                             double   *onset_sub_off_out)
316     {
317         if (!d->env_ring || d->env_ring_capacity == 0) return 0;
318         if (c_peak_idx == 0)                            return 0;
319     
320         double threshold = 0.5 * c_peak_value;
321         if (threshold <= 0.0) return 0;
322     
323         uint64_t search_max = d->c_onset_search_max_samples;
324         if (search_max == 0) return 0;
325     
326         /* Don't walk past where the burst started + skip window: that's
327          * by definition before the C-search region. */
328         uint64_t earliest_idx;
329         {
330             uint64_t skip_end = d->burst_start_idx + d->c_search_skip_samples;
331             uint64_t window_lo = (c_peak_idx > search_max)
332                                      ? (c_peak_idx - search_max) : 0;
333             earliest_idx = (skip_end > window_lo) ? skip_end : window_lo;
334         }
335         /* Also bounded by what the ring buffer holds. */
336         {
337             uint64_t ring_oldest = (d->env_ring_newest_abs >= d->env_ring_capacity - 1)
338                                        ? (d->env_ring_newest_abs - (d->env_ring_capacity - 1))
339                                        : 0;
340             if (earliest_idx < ring_oldest) earliest_idx = ring_oldest;
341         }
342         if (earliest_idx >= c_peak_idx) return 0;
343     
344         /* Walk back: i is the candidate sample index. Track consecutive
345          * below-threshold samples. The "last >= threshold" tracker holds
346          * the index of the most recent sample we've seen with env >= thr,
347          * which becomes the onset boundary when dwell is reached. */
348         uint64_t consecutive_below = 0;
349         int      have_above        = 0;
350         uint64_t last_above_idx    = 0;
351         double   last_above_val    = 0.0;
352     
353         /* Start from one sample earlier than the peak (peak itself is
354          * above threshold by construction). Walk down toward earliest_idx. */
355         for (uint64_t i = c_peak_idx; i > earliest_idx; ) {
356             --i;
357             double v = env_ring_at(d, i);
358             if (v >= threshold) {
359                 have_above        = 1;
360                 last_above_idx    = i;
361                 last_above_val    = v;
362                 consecutive_below = 0;
363             } else {
364                 consecutive_below++;
365                 if (have_above && consecutive_below >= d->c_onset_dwell_samples) {
366                     /* Onset is the threshold crossing between the sample
367                      * just before last_above_idx (env < thr) and
368                      * last_above_idx (env >= thr). Linear interp on
369                      * threshold-crossing. */
370                     double v_prev = (last_above_idx > 0)
371                                         ? env_ring_at(d, last_above_idx - 1)
372                                         : 0.0;
373                     double frac = 0.0;
374                     if (last_above_val > v_prev) {
375                         /* monotonic rise across the crossing */
376                         frac = (threshold - v_prev) / (last_above_val - v_prev);
377                         if (frac < 0.0) frac = 0.0;
378                         if (frac > 1.0) frac = 1.0;
379                     }
380                     /* sample_index convention: integer index just at or
381                      * after the crossing; sub_off in [-0.5, +0.5]. */
382                     *onset_idx_out     = last_above_idx;
383                     *onset_sub_off_out = frac - 1.0;  /* in [-1, 0] */
384                     /* Renormalize to canonical [-0.5, +0.5] range by
385                      * shifting integer index when fraction < -0.5. */
386                     if (*onset_sub_off_out < -0.5) {
387                         *onset_sub_off_out += 1.0;
388                         if (*onset_idx_out > 0) (*onset_idx_out)--;
389                     }
390                     return 1;
391                 }
392             }
393         }
394         /* Reached the search bound without finding a clean dwell. */
395         return 0;
396     }

398     /* Insertion-sort a small array in place. N <= 16 so this is fine. */
399     static void insertion_sort(double *a, int n) {
400         for (int i = 1; i < n; ++i) {
401             double x = a[i];
402             int j = i - 1;
403             while (j >= 0 && a[j] > x) { a[j+1] = a[j]; --j; }
404             a[j+1] = x;
405         }
406     }

408     /* Recompute cached median after inserting a peak. Called infrequently. */
409     static void update_median_cache(tg_detector_t *d) {
410         int n = d->peak_history_count;
411         if (n < 1) { d->median_peak_cache = 0.0; return; }
412         double tmp[TG_PEAK_HISTORY_N];
413         for (int i = 0; i < n; ++i) tmp[i] = d->peak_history[i];
414         insertion_sort(tmp, n);
415         if (n & 1) {
416             d->median_peak_cache = tmp[n/2];
417         } else {
418             d->median_peak_cache = 0.5 * (tmp[n/2 - 1] + tmp[n/2]);
419         }
420     }

422     /* V5.6: regime-change detector helpers.
423      *
424      * The regime ring tracks recent burst peak values. When a new peak
425      * arrives that's at least TG_REGIME_RATIO (10x) larger than the
426      * minimum of the ring, we conclude the acoustic regime has changed
427      * (e.g. quiet ambient noise -> watch on microphone) and trip a
428      * reset request. The library checks the reset_pending flag after
429      * each detector_process and, if set, flushes adaptive state.
430      *
431      * Suppression gates:
432      *   - Need >= 4 prior peaks in the ring before considering a trip
433      *     (otherwise the first few peaks of a fresh stream would always
434      *     trip on themselves).
435      *   - Both old min and new peak must be above TG_REGIME_FLOOR
436      *     (default 0.001). Below that we're comparing noise to noise.
437      *   - Cooldown: TG_REGIME_COOLDOWN_S (1 s) between resets, tracked
438      *     by sample-index distance. Stops consecutive loud impulses
439      *     from each triggering their own reset and preventing sync
440      *     from ever settling.
441      */
442     static double regime_ring_min(const tg_detector_t *d) {
443         if (d->regime_peak_count <= 0) return 0.0;
444         double m = d->regime_peak_ring[0];
445         for (int i = 1; i < d->regime_peak_count; ++i) {
446             if (d->regime_peak_ring[i] < m) m = d->regime_peak_ring[i];
447         }
448         return m;
449     }

451     static void regime_push_peak(tg_detector_t *d, double peak, uint64_t abs_idx) {
452         /* Check for regime trip BEFORE storing the new peak: we want to
453          * compare against the prior baseline, not include the trigger in
454          * the comparison. */
455         if (d->regime_peak_count >= 4) {
456             double prev_min = regime_ring_min(d);
457             int above_floor = (peak       >= TG_REGIME_FLOOR &&
458                                prev_min   >= TG_REGIME_FLOOR);
459             int abs_floor_jump = (peak >= TG_REGIME_FLOOR &&
460                                   prev_min < TG_REGIME_FLOOR);
461             /* Trip if both peaks are above the floor and the ratio is
462              * large enough, OR if the new peak is above the floor and
463              * the old min was below it (jump from noise to signal). */
464             int ratio_ok = above_floor && (peak >= TG_REGIME_RATIO * prev_min);
465             int trip = ratio_ok || abs_floor_jump;
466             /* Apply cooldown: skip the trip if we recently reset. */
467             if (trip && d->regime_last_reset_idx > 0) {
468                 uint64_t since = abs_idx - d->regime_last_reset_idx;
469                 uint64_t cooldown = (uint64_t)(TG_REGIME_COOLDOWN_S * d->fs);
470                 if (since < cooldown) trip = 0;
471             }
472             if (trip) {
473                 d->regime_reset_pending = 1;
474                 d->regime_last_reset_idx = abs_idx;
475             }
476         }
477         /* Always store the new peak into the ring (regardless of trip). */
478         d->regime_peak_ring[d->regime_peak_head] = peak;
479         d->regime_peak_head = (d->regime_peak_head + 1) % TG_REGIME_RING_N;
480         if (d->regime_peak_count < TG_REGIME_RING_N) d->regime_peak_count++;
481     }

483     /* Push a newly-observed peak into the ring buffer and refresh median.
484      * V5.6: also feed the regime-change detector. */
485     static void push_peak(tg_detector_t *d, double peak, uint64_t abs_idx) {
486         d->peak_history[d->peak_history_head] = peak;
487         d->peak_history_head = (d->peak_history_head + 1) % TG_PEAK_HISTORY_N;
488         if (d->peak_history_count < TG_PEAK_HISTORY_N) d->peak_history_count++;
489         update_median_cache(d);
490         regime_push_peak(d, peak, abs_idx);
491     }

493     /* Compute 75th percentile of noise_history using partial sort. */
494     static void update_noise_percentile_cache(tg_detector_t *d) {
495         int n = d->noise_history_count;
496         if (n < 1) { d->noise_percentile_cache = 0.0; return; }
497         /* Copy & sort. N <= 256, O(N^2) insertion sort is fine (~65k ops
498          * worst case, once per millisecond of silence). */
499         double tmp[TG_NOISE_HISTORY_N];
500         for (int i = 0; i < n; ++i) tmp[i] = d->noise_history[i];
501         insertion_sort(tmp, n);
502         /* 75th percentile index */
503         int idx = (3 * (n - 1)) / 4;
504         d->noise_percentile_cache = tmp[idx];
505     }
506     
507     /* Record one downsampled silence-region envelope sample. Called at most
508      * once per d->noise_sample_interval samples, only when not in_burst. */
509     static void push_noise_sample(tg_detector_t *d, double e) {
510         d->noise_history[d->noise_history_head] = e;
511         d->noise_history_head = (d->noise_history_head + 1) % TG_NOISE_HISTORY_N;
512         if (d->noise_history_count < TG_NOISE_HISTORY_N) d->noise_history_count++;
513         update_noise_percentile_cache(d);
514     }
515     
516     /* Return the current "effective noise floor" for threshold computation.
517      *
518      * Bootstrap behavior: until we have at least 32 silence samples in the
519      * history (~32 ms), fall back to the EMA-tracked min-like noise floor in
520      * d->noise_floor. Once the percentile buffer has enough data, use the
521      * 75th percentile instead -- this tracks the typical ambient level
522      * between ticks rather than its minimum, so the onset threshold sits
523      * ABOVE ambient noise instead of within it. */
524     static double effective_noise(const tg_detector_t *d) {
525         if (d->noise_history_count < 32) {
526             return d->noise_floor;
527         }
528         /* Floor the percentile at the EMA noise to avoid pathology if
529          * history gets filled with zeros during warmup. */
530         double p = d->noise_percentile_cache;
531         double f = d->noise_floor;
532         return (p > f) ? p : f;
533     }
534     
535     /* Compute the current reference peak level used to derive thresholds.
536      *
537      * Phases:
538      *   1. No peaks yet (count == 0):      use signal_ceiling (max-hold).
539      *      Needed so initial thresholds are meaningful before any peak has
540      *      been committed to history.
541      *   2. Bootstrapping (count 1..3):     use max of peak_history.
542      *      Avoids median-of-small-sample bias; still robust to single
543      *      spurious-first-burst as we've accepted at most a few.
544      *   3. Normal operation (count >= 4):  use median of peak_history.
545      *      Robust to outliers and sparse-period drift.
546      *
547      * Always clamped to at least 10 * noise_floor so the threshold calc
548      * doesn't collapse below the noise envelope. */
549     static double reference_peak(const tg_detector_t *d) {
550         double floor_ref = 10.0 * d->noise_floor;
551         double r;
552         if (d->peak_history_count == 0) {
553             r = d->signal_ceiling;
554         } else if (d->peak_history_count < 4) {
555             r = 0.0;
556             for (int i = 0; i < d->peak_history_count; ++i) {
557                 if (d->peak_history[i] > r) r = d->peak_history[i];
558             }
559         } else {
560             r = d->median_peak_cache;
561         }
562         return (r > floor_ref) ? r : floor_ref;
563     }
564     
565     /* Compute the current thresholds from the noise floor and reference
566      * peak. Used both inside the main loop and by the public getter. */
567     static void compute_thresholds(const tg_detector_t *d,
568                                    double *eff_noise,
569                                    double *ref_peak,
570                                    double *span,
571                                    double *onset_thr,
572                                    double *min_peak_thr)
573     {
574         double n  = effective_noise(d);
575         double r  = reference_peak(d);
576         double sp = r - n;
577         if (sp < n * 2.0) sp = n * 2.0;
578         if (eff_noise)    *eff_noise    = n;
579         if (ref_peak)     *ref_peak     = r;
580         if (span)         *span         = sp;
581         if (onset_thr)    *onset_thr    = n + d->onset_fraction    * sp;
582         if (min_peak_thr) *min_peak_thr = n + d->min_peak_fraction * sp;
583     }
584     
585     void tg_detector_get_thresholds(const tg_detector_t *d,
586                                     double *onset_thr,
587                                     double *min_peak_thr,
588                                     double *eff_noise,
589                                     double *ref_peak)
590     {
591         compute_thresholds(d, eff_noise, ref_peak, NULL, onset_thr, min_peak_thr);
592     }

616     size_t tg_detector_process(tg_detector_t *d,
617                                const float *envelope, size_t n,
618                                tg_raw_event_t *out_events,
619                                size_t *out_count, size_t max_events)
620     {
621         size_t produced = 0;
622     
623         for (size_t i = 0; i < n; ++i) {
624             double e = (double)envelope[i];
625             uint64_t abs_idx = d->total_samples + i;
626     
627             /* V5.4: write current envelope sample into the ring buffer.
628              * Used by the C-onset backward walk at burst-emission time. */
629             if (d->env_ring && d->env_ring_capacity > 0) {
630                 d->env_ring[d->env_ring_head] = (float)e;
631                 d->env_ring_head = (d->env_ring_head + 1) % d->env_ring_capacity;
632                 d->env_ring_newest_abs = abs_idx;
633                 d->env_ring_has_data   = 1;
634             }
635     
636             /* EMA-tracked noise floor: snap-down min-tracker, used only as
637              * bootstrap fallback before the percentile buffer fills. */
638             if (!d->in_burst) {
639                 if (e < d->noise_floor) {
640                     d->noise_floor = e;
641                 } else {
642                     d->noise_floor += d->noise_alpha * (e - d->noise_floor);
643                 }
644                 if (d->noise_floor < 1e-9) d->noise_floor = 1e-9;
645     
646                 /* Record one silence sample per ~1ms into the noise percentile
647                  * history. These are our observations of the typical ambient
648                  * level between ticks. Only collected during silence so burst
649                  * energy doesn't leak in. */
650                 if (abs_idx >= d->noise_last_sample_idx + d->noise_sample_interval) {
651                     push_noise_sample(d, e);
652                     d->noise_last_sample_idx = abs_idx;
653                 }
654             }
655     
656             /* Keep the legacy signal_ceiling updated (for diagnostics). */
657             if (e > d->signal_ceiling) {
658                 d->signal_ceiling = e;
659             } else {
660                 d->signal_ceiling += d->ceil_alpha * (e - d->signal_ceiling);
661             }
662             if (d->signal_ceiling < d->noise_floor * 3.0) {
663                 d->signal_ceiling = d->noise_floor * 3.0;
664             }
665     
666             if (abs_idx < d->warmup_samples) {
667                 d->prev_sample = e;
668                 continue;
669             }
670     
671             /* Thresholds anchored to median of recent peak heights.
672              *
673              * reference_peak() returns either:
674              *   - 10 * noise_floor (no history yet)
675              *   - 0.3 * recent_max (1..3 peaks: bootstrapping)
676              *   - median of last N peaks (>=4 peaks: normal operation)
677              *
678              * Unlike the old max-hold ceiling, this is robust to both single
679              * loud outliers (median doesn't move) and long quiet stretches
680              * (median doesn't decay). After a loud bang, the onset threshold
681              * stays anchored to the typical tick peak, not the bang, so
682              * subsequent quieter real ticks still clear it. After a gap with
683              * no ticks, the reference stays at the typical tick level so
684              * background noise can't cross the onset threshold.
685              *
686              * Onset = 3% above effective_noise : catches true silence->signal.
687              *         Because effective_noise is the 75th-percentile of recent
688              *         silence samples (not the minimum), this sits ABOVE
689              *         typical ambient noise instead of inside it. So 3% is
690              *         enough headroom to catch real ticks reliably without
691              *         tripping on background rumble.
692              * Min-peak = 20% of dynamic range   : rejects noise bumps */
693             double eff_noise, ref_peak, span, onset_thr, min_peak;
694             compute_thresholds(d, &eff_noise, &ref_peak, &span,
695                                &onset_thr, &min_peak);
696     
697             /* Silence gate is now wall-clock based: time since the last burst
698              * ended, not continuous-below-release count. The purpose of the
699              * gate is "enough time has passed for this to be a new beat",
700              * which is independent of whether the envelope has continuously
701              * stayed below a threshold. On watches with long ring-down tails
702              * (NH35) the envelope can hover near release_thr for many ms
703              * after the impact, which would reset a count-based gate and
704              * block the next legitimate tick.
705              *
706              * We increment silence_samples every sample we're NOT in a burst,
707              * unconditionally -- it represents "samples since last burst
708              * ended", i.e. time elapsed. */
709             if (!d->in_burst) {
710                 d->silence_samples++;
711             }
712             /* V4.5: count wall-clock samples since last A onset, regardless
713              * of in-burst state. Used to gate sub-impulses that arrive
714              * inside a multi-impulse beat cluster. Saturate at a large
715              * value to avoid overflow on long silent stretches. */
716             if (d->samples_since_last_a < (1ULL << 60)) {
717                 d->samples_since_last_a++;
718             }
719     
720             if (!d->in_burst) {
721                 /* Silence -> burst transition.
722                  * Two gates that must BOTH pass:
723                  *   1) silence_samples >= min_silence_samples
724                  *      (time since previous burst ended)
725                  *   2) samples_since_last_a >= min_a_interval_samples
726                  *      (time since previous A onset; V4.5 sub-impulse gate)
727                  *
728                  * Pre-sync min_a_interval_samples is 0 (gate disabled, so
729                  * BPH detection sees all candidate events). Once sync is
730                  * locked, the library sets this to ~70% of beat_period to
731                  * suppress sub-impulses inside multi-impulse beat clusters. */
732                 uint64_t min_silence_samples = d->min_silence_samples;
733                 if (e > onset_thr && d->prev_sample <= onset_thr &&
734                     d->silence_samples >= min_silence_samples &&
735                     d->samples_since_last_a >= d->min_a_interval_samples)
736                 {
737                     d->in_burst = 1;
738                     d->silence_samples = 0;
739                     /* NOTE: samples_since_last_a is NOT reset here. The
740                      * burst may still fail the min-peak test at burst-end,
741                      * in which case no A is emitted and the counter must
742                      * not be reset. Reset happens after successful A emit. */
743     
744                     /* Sub-sample crossing time: linear interpolation between
745                      * prev_sample (below threshold) and current sample (above). */
746                     double frac = 0.0;
747                     double denom = e - d->prev_sample;
748                     if (denom > 1e-20) {
749                         frac = (onset_thr - d->prev_sample) / denom;
750                         if (frac < 0.0) frac = 0.0;
751                         if (frac > 1.0) frac = 1.0;
752                     }
753                     /* Crossing is at time (abs_idx - 1 + frac). Encode as
754                      * integer sample index + offset in [-0.5, +0.5]. */
755                     uint64_t idx = abs_idx - 1;
756                     double sub = frac;
757                     if (sub > 0.5) { idx += 1; sub -= 1.0; }
758                     d->burst_start_idx    = idx;
759                     d->burst_start_offset = sub;
760                     d->burst_start_time   = ((double)idx + sub) / d->fs;
761     
762                     /* Reset burst max tracking (height check / fallback for C) */
763                     d->burst_max          = e;
764                     d->burst_max_idx      = abs_idx;
765                     d->burst_max_y_minus1 = d->prev_sample;
766                     d->have_peak_plus1    = 0;
767     
768                     /* V5.2: reset C-peak tracker. C-peak only updates after
769                      * c_search_skip_samples have elapsed since burst start. */
770                     d->c_have_peak        = 0;
771                     d->burst_c_max        = 0.0;
772                     d->burst_c_idx        = 0;
773                     d->c_have_peak_plus1  = 0;
774     
775                     /* DON'T emit A yet -- defer until burst ends and we know
776                      * the peak height. This lets us discard noise bumps that
777                      * cross the low onset threshold but never reach a real
778                      * peak. A is emitted at burst-end time using the saved
779                      * burst_start_* fields. */
780                 }
781             } else {
782                 /* Inside a burst. Track running max for the C event.
783                  *
784                  * Burst-end detection is WALL-CLOCK based: enough time since
785                  * the peak was observed. Wall-clock rather than "continuous
786                  * below release" because long ring-down tails oscillate
787                  * around release threshold without ever continuously staying
788                  * below for the required duration -- that would trap us
789                  * inside a burst indefinitely and miss all subsequent beats.
790                  * Once the peak has happened and enough time has passed, the
791                  * tick is effectively over. If a higher sample appears, the
792                  * peak is updated and the timer resets. */
793                 if (e > d->burst_max) {
794                     d->burst_max          = e;
795                     d->burst_max_idx      = abs_idx;
796                     d->burst_max_y_minus1 = d->prev_sample;
797                     d->have_peak_plus1    = 0;
798                 } else if (!d->have_peak_plus1 && abs_idx == d->burst_max_idx + 1) {
799                     d->burst_max_y_plus1  = e;
800                     d->have_peak_plus1    = 1;
801                 }
802     
803                 /* V5.2: independently track C-peak only past the skip
804                  * window (counted from burst start). burst_max may have
805                  * latched on a sub-impulse near burst start; the C-peak
806                  * tracker ignores those and finds the true C cluster
807                  * peak. */
808                 uint64_t samples_since_burst = abs_idx - d->burst_start_idx;
809                 if (samples_since_burst >= d->c_search_skip_samples) {
810                     if (!d->c_have_peak || e > d->burst_c_max) {
811                         d->burst_c_max        = e;
812                         d->burst_c_idx        = abs_idx;
813                         d->burst_c_y_minus1   = d->prev_sample;
814                         d->c_have_peak        = 1;
815                         d->c_have_peak_plus1  = 0;
816                     } else if (!d->c_have_peak_plus1 && abs_idx == d->burst_c_idx + 1) {
817                         d->burst_c_y_plus1    = e;
818                         d->c_have_peak_plus1  = 1;
819                     }
820                 }
821     
822                 uint64_t samples_since_peak = abs_idx - d->burst_max_idx;
823     
824                 if (samples_since_peak >= d->burst_end_samples) {
825                     /* Minimum tick height: fraction of the dynamic range. A
826                      * real tick's peak is typically at or near reference_peak,
827                      * so 20% (default) is well below what we'd expect but well
828                      * above any noise-bump that crossed the onset threshold.
829                      * Adjustable via tg_detector_set_min_peak_fraction(). */
830                     double min_peak = effective_noise(d) + d->min_peak_fraction * span;
831                     int is_real_tick = (d->burst_max >= min_peak);
832     
833                     if (is_real_tick) {
834                         /* Emit A first (at burst start onset time) */
835                         if (produced + *out_count < max_events) {
836                             tg_raw_event_t *ev = &out_events[*out_count + produced];
837                             ev->sample_index      = d->burst_start_idx;
838                             ev->sub_sample_offset = d->burst_start_offset;
839                             ev->time_seconds      = d->burst_start_time;
840                             ev->peak_value        = (float)d->burst_max;
841                             ev->is_onset          = 1;
842                             /* V5.4: A events don't carry onset metadata
843                              * (the A's primary timing already IS the onset). */
844                             ev->onset_sample_index      = 0;
845                             ev->onset_sub_sample_offset = 0.0;
846                             ev->onset_time_seconds      = 0.0;
847                             ev->onset_valid             = 0;
848                             produced++;
849                         }
850                         /* V4.5: now that an A has actually been emitted,
851                          * reset the A-to-A interval counter. Failed bursts
852                          * (above) do NOT reset it -- they're noise that
853                          * passed the onset gate but failed min-peak.
854                          *
855                          * BUG-FIX: the counter must measure A-to-A, not
856                          * burst-end-to-next-A. We're at burst-end now (which
857                          * is ~25 ms after the C peak, ~30 ms after the A
858                          * onset for typical 28800 BPH). If we reset to 0
859                          * here, the next A is gated by 0.7*period from
860                          * THIS moment, which effectively requires
861                          * (0.7 * period + ~30 ms) between A onsets. For
862                          * watches with high beat error (alternating long
863                          * and short A-to-A intervals), the short ones can
864                          * be just barely too short -- the small first
865                          * impulse gets missed and A is mis-placed at the
866                          * tail or on the C onset.
867                          *
868                          * Correct: reset to (current - burst_start) so
869                          * the counter reflects time since A onset. */
870                         uint64_t burst_dur_samples = abs_idx - d->burst_start_idx;
871                         d->samples_since_last_a = burst_dur_samples;
872                         /* V5.2: pick C-peak (post-skip) if available; else
873                          * fall back to burst_max for backward compat (very
874                          * short bursts, or skip > burst duration). */
875                         double   c_max_v;
876                         uint64_t c_max_idx;
877                         double   c_y_m1, c_y_p1;
878                         int      c_have_p1;
879                         if (d->c_have_peak) {
880                             c_max_v   = d->burst_c_max;
881                             c_max_idx = d->burst_c_idx;
882                             c_y_m1    = d->burst_c_y_minus1;
883                             c_y_p1    = d->burst_c_y_plus1;
884                             c_have_p1 = d->c_have_peak_plus1;
885                         } else {
886                             c_max_v   = d->burst_max;
887                             c_max_idx = d->burst_max_idx;
888                             c_y_m1    = d->burst_max_y_minus1;
889                             c_y_p1    = d->burst_max_y_plus1;
890                             c_have_p1 = d->have_peak_plus1;
891                         }
892     
893                         /* Emit C (at burst peak, with parabolic interp) */
894                         double sub_off = 0.0;
895                         if (c_have_p1) {
896                             sub_off = parabolic_offset(c_y_m1, c_max_v, c_y_p1);
897                         }
898     
899                         /* V5.4: find C-cluster onset by walking back from the
900                          * C peak in the recent envelope ring buffer. Always
901                          * computed (cheap); the public-API layer chooses
902                          * whether to promote it to primary timing or keep
903                          * it as metadata. */
904                         uint64_t onset_idx     = 0;
905                         double   onset_sub_off = 0.0;
906                         int      onset_valid   =
907                             find_c_onset(d, c_max_idx, c_max_v,
908                                          &onset_idx, &onset_sub_off);
909     
910                         if (produced + *out_count < max_events) {
911                             tg_raw_event_t *ev = &out_events[*out_count + produced];
912                             ev->sample_index      = c_max_idx;
913                             ev->sub_sample_offset = sub_off;
914                             ev->time_seconds      = ((double)c_max_idx + sub_off) / d->fs;
915                             ev->peak_value        = (float)c_max_v;
916                             ev->is_onset          = 0;
917                             /* V5.4: onset metadata */
918                             ev->onset_sample_index      = onset_idx;
919                             ev->onset_sub_sample_offset = onset_sub_off;
920                             ev->onset_time_seconds      =
921                                 onset_valid
922                                     ? ((double)onset_idx + onset_sub_off) / d->fs
923                                     : 0.0;
924                             ev->onset_valid             = onset_valid;
925                             produced++;
926                         }
927                         d->last_event_sample   = c_max_idx;
928                         /* Credit the wall-clock time since peak toward the
929                          * next silence gate. Use samples_since_peak (from
930                          * burst_max_idx) -- this preserves V5.1 behavior
931                          * for the silence-gate sizing, even when C is
932                          * placed at a different (later) sample via the
933                          * V5.2 skip mechanism. */
934                         d->silence_samples     = samples_since_peak;
935                         /* Record this real peak in the height history. Use
936                          * burst_max (the loudest sample in the burst) so
937                          * peak history reflects burst loudness independent
938                          * of where C was placed. abs_idx for V5.6 regime
939                          * detector is the current sample index. */
940                         push_peak(d, d->burst_max, abs_idx);
941                     } else {
942                         /* Noise bump rejected - don't penalize next real tick
943                          * by zeroing silence counter. Pretend the whole episode
944                          * was quiet so the silence-gate is open for next onset.
945                          * Do NOT add to peak history -- history should only
946                          * contain real ticks. */
947                         d->silence_samples = 1000000;
948                     }
949                     d->in_burst          = 0;
950                 }
951             }
952     
953             d->prev_sample = e;
954         }
955     
956         d->total_samples += n;
957         *out_count += produced;
958         return produced;
959     }

961     /* V5.6: atomic read-and-clear of the regime reset flag. Called by
962      * the library (Timegrapher.cpp) after each tg_detector_process to
963      * see if the detector observed a regime change during that batch.
964      * Returns 1 once per detected change; subsequent calls return 0
965      * until the next change. */
